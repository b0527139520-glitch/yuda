רבי יהודה בר אלעאי והלחמנייה שבבוץ

👨‍🏫 כולל מדריך למורה
  • דְּמוּת הַגִּבּוֹר: הַתַּנָּא הַקָּדוֹשׁ רַבִּי יְהוּדָה בַּר אִלְעָאִי, מִגְּדוֹלֵי תַּלְמִידָיו שֶׁל רַבִּי עֲקִיבָא, הַמֻּכָּר בַּשַּׁבָּחָה הָעֲצוּמָה שֶׁלּוֹ כְּ"רֹאשׁ הַמְדַבְּרִים בְּכָל מָקוֹם" וּבַעֲנִיוּתוֹ הַמֻּפְלָאָה.
  • הַקּוֹנְפְלִיקְט: הַפַּעַר הָעָצוּם בֵּין הַזִּלְזוּל שֶׁל יְלָדִים הַמְשׂัחֲקִים בְּלַחְמָנִיָּה כִּבְכַדּוּר וּמַשְׁלִיכִים אוֹתָהּ לַבּוֹץ, לְבֵין הָעָמָל הַגּוּפָנִי וְהָרוּחָנִי הַנִּדְרָשׁ לִיצִירַת כָּל פְּרוּסַת לֶחֶם בָּעוֹלָם שֶׁנּוֹצְרָה בְּמַתְּנַת שָׁמַיִם.
  • נְקֻדַּת הַשִּׂיא: רַבִּי יְהוּדָה, גְּדוֹל הַדּוֹר, מִתְכּוֹפֵף בְּמוֹ יָדָיו אֶל הַבּוֹץ הַלַּח וְהַשָּׁחֹר, מֵרִים אֶת הַלַּחְמָנִיָּה הַמְלֻכְלֶכֶת, מְנַקֶּה אוֹתָהּ בַּעֲדִינוּת וּמְנַשֵּׁק אוֹתָהּ בִּדְמָעוֹת שֶׁל אַהֲבָה וְהַעֲרָכָה לַבּוֹרֵא יִתְבָּרַךְ.
  • מוּסַר הַשְּׂכֵל: כָּל פְּרוּסַת לֶחֶם וּמַאֲכָל שֶׁנִּכְנָסִים לְפִינוּ הֵם אוֹצָר קָדוֹשׁ שֶׁנִּבְרָא בְּאַהֲבָת ה' וּבְעָמָל רַב. אָסוּר לְזַלְזֵל בּוֹ אוֹ לְהַשְׁלִיכוֹ בְּחֻסֶּר כָּבוֹד, וְעָלֵינוּ לִשְׂמֹחַ בַּחֵלֶק שֶׁלָּנוּ וּלְהוֹדוֹת עָלָיו בְּלֵב שָׁלֵם.

## הַסִּפּוּר הַמָּלֵא: רַבִּי יְהוּדָה וְהַלַּחְמַּנִיָּה שֶׁבַּבּוֹץ

🔹 א. הַמְּעִיל הַמְּשֻׁתָּף וְהַפָּנִים הַמְּאִירוֹת

הַקֹּר בְּחוּצוֹת הָעִיר אוּשָׁא שֶׁבַּגָּלִיל הָיָה עַז וְדּוֹקֵר. הָרוּחַ הַחֲזָקָה שָׁרְקָה בֵּין בָּתֵּי הָאֶבֶן הַפְּשׁוּטִים, מַרְעִידָה אֶת עַנְפֵי עֲצֵי הַזַּיִת הָעַתִּיקִים שֶׁעָמְדוּ בְּצִדֵּי הַדְּרָכִים. בְּתוֹךְ בַּיִת קָטָן וְדַּל מֵאֲבָנִים וְטִיט, יָשַׁב הַתַּנָּא הַקָּדוֹשׁ רַבִּי יְהוּדָה בַּר אִלְעָאִי.

הַבַּיִת הָיָה כִּמְעַט רֵיק לְגַמְרֵי. לֹא הָיוּ בּוֹ כִּסֵּאות רַכִּים אוֹ שֻׁלְחָנוֹת מְפֹאָרִים. בְּקָצֶה אֶחָד עָמְדוּ שְׁנֵי סַפְסְלֵי עֵץ יְשָׁנִים, וּבַקָּצֶה הַשֵּׁנִי מִטָּה דַּלָּה. רַבִּי יְהוּדָה וְאִשְׁתּוֹ הַצַּדֶּקֶת חָיוּ בְּעֹנִי כֹּה עָצוּם, עַד שֶׁבְּכָל בֵּיתָם לֹא הָיָה אֶלָּא מְעִיל חַם וְעָבֶה אֶחָד בִּלְבַד! כַּאֲשֶׁר הָיְתָה הָרַבָּנִית צְרִיכָה לָצֵאת אֶל הַשּׁוּק לִקְנוֹת מְעַט יְרָקוֹת, הָיְתָה לוֹבֶשֶׁת אֶת הַמְּעִיל, וְרַבִּי יְהוּדָה הָיָה נִשְׁאָר בַּבַּיִת וְלוֹמֵד תּוֹרָה. וְכַאֲשֶׁר הָיָה הָרַב צָרִיךְ לָלֶכֶת אֶל בֵּית הַמִּדְרָשׁ לְלַמֵּד אֶת תַּלְמִּידָיו, הָיָה הוּא לוֹבֵשׁ אֶת הַמְּעִיל, וְאִשְׁתּוֹ נִשְׁאֶרֶת בַּבַּיִת.

לַמְּרוֹת הָעֹנִי הַכָּבֵד וְהָרָעָב שֶׁהָיָה מְקַרְקֵר לִפְעָמִּים בְּבִטְנוֹ, פָּנָיו שֶׁל רַבִּי יְהוּדָה הָיוּ תָּמִּיד צְהֻבּוֹת וּמְּאִירוֹת בְּאוֹר מֻפְלָא! כַּאֲשֶׁר עָבַר גּוֹי אֶחָד בָּרְחוֹב וְרָאָה אֶת פָּנָיו הַקּוֹרְנוֹת שֶׁל הָרַב, צָחַק וְאָמַּר: "הַהוּא אָדָם אוֹ שֶׁהוּא שׁוֹתֶה יַיִן יָקָר כָּל הַיּוֹם, אוֹ שֶׁהוּא עָשִׁיר גָּדוֹל!". שָׁמַּע זאת רַבִּי יְהוּדָה וְחִיֵּךְ בַּעֲנָוָה: "חָלִילָה! אֵינִי שׁוֹתֶה יַיִן אֶלָּא מִכּוֹס אַרְבַּע כּוֹסוֹת שֶׁל פֶּסַח. וְהָאוֹר שֶׁבְּפָנַי מַגִּיעַ רַק מִּמְּתוּקוֹתָהּ שֶׁל הַתּוֹרָה, שֶׁנֶּאֱמַּר: 'חָכְמַּת אָדָם תָּאִיר פָּנָיו'!".

בַּבֹּקֶר הַהוּא, אָכַל רַבִּי יְהוּדָה אֲרוּחָה דַּלָּה וּפְשׁוּטָה מְאֹד – מְעַט סֶלֶק יָבֵשׁ וּמַיִם קָרִים. הוּא בֵּרֵךְ בִּבְרָכָה רַבָּה וּבְכַוָּנָה עֲצוּמָּה לַה' יִתְבָּרַךְ עַל כָּל נְשִׁימָּה וּנְשִׁימָּה, לָבַשׁ אֶת הַמְּעִיל הַמְּשֻׁתָּף, וְיָצָא אֶל סִמְּטָאוֹת הָעִיר הַבּוּצִיּוֹת בְּדַרְכּוֹ לְהַרְבִּיץ תּוֹרָה בַּבֵּית הַמִּדְרָשׁ.

🔹 ב. קוֹלוֹת צְחוֹק בַּסִּמְּטָה הַרְטֻבָּה

הַגְּשָׁמִּים הַכְּבֵדִים שֶׁיָּרְדוּ בַּלַּיְלָה הָפְכוּ אֶת הַשְּׁבִילִים שֶׁל אוּשָׁא לִשְׁלוּלִיּוֹת עֲמוּקֹות שֶׁל בּוֹץ שָׁחֹר וְדָּבִיק. רַבִּי יְהוּדָה צָעַד לְאִטּוֹ, נִזְהָר שֶׁלֹּא לְהַשְׁחִית אוֹ לְלַכְלֵךְ אֶת בְּגָדָיו הַפְּשׁוּטִים, כְּשֶׁלְּפֶתַע נִשְׁמְּעוּ קוֹלוֹת שֶׁל צְחוֹק, צְעָקוֹת עַלִּיזוֹת וּרְעָשִׁים שֶׁל הִתְקַטְּגוּת וּמִּשְׂחָק מִּקְּצֵה הָרְחוֹב.

הוּא הִבִּיט וְרָאָה קְבוּצָה שֶׁל בָּנִים, בְּעֵרֶךְ בְּגִילֵי שְׁמוֹנֶה וְתֵשַׁע, הַמְּשַׂחֲקִים בִּסְעָרָה בְּאֶמְּצָע הַדֶּרֶךְ. הַבָּנִים הָיוּ אֲדֻמֵּי לְחָיַיִם מִּן הַמִּשְׂחָק, בְּגָדֵיהֶם הָיוּ מְּלֻכְלָכִים מִּן הַבּוֹץ, וְהֵם רָצוּ וְקָפְצוּ מִצַּד לְצַד. אַךְ כַּאֲשֶׁר הִבְחִין רַבִּי יְהוּדָה בַּ"כַּדּוּר" שֶׁבּוֹ הֵם מְשַׂחֲקִים – נֶחְרַד לִבּוֹ הַצַּדִּיק! הַיְּלָדִים לא שִׂמְּחוּ בְּכַדּוּר רגִיל. בִּידֵיהֶם הָיְתָה לַחְמַּנִיָּה עֲגֻלָּה, טְרִיָּה וּזְהֻבָּה, שֶׁנֶּאֶפְתָה זֶה עַתָּה בַּתַּנּוּר וְרֵיחָהּ הַמָּתוֹק וְהַנִּפְלָא עוֹד עָמַּד בָּאֲוִיר. אֶחָד הַיְּלָדִים זָרַק אֶת הַלַּחְמַּנִיָּה בָּאֲוִיר, וְהַשֵּׁנִי חָבַט בָּהּ בְּכֹחַ בְּרַגְלוֹ! הַלַּחְמַּנִיָּה הַקְּדוֹשָׁה עָפָה בַּקֶּשֶׁת, וּבִתְרוּעַת צְחוֹק שֶׁל הַבָּנִים – נָפְלָה וְהִתְרַסְּקָה יְשִׁירוֹת אֶל תּוֹךְ שְׁלוּלִית עֲמוּקָה שֶׁל בּוֹץ שָׁחֹר וּמְּלֻכְלָךְ.

"הַצְלָחָה!" קָרָא אֶחָד הַיְּלָדִים, וְנִגַּשׁ לִבְעֹט בַּלַּחְמַּנִיָּה הַשְּׁקוּעָה בַּבּוֹץ כְּדֵי לְהַמְשִׁיךְ בַּמִּשְׂחָק הַמְבַזֶּה.

🔹 ג. הַתַּנָּא הַקָּדוֹשׁ מִתְכּוֹפֵף אֶל הָאֲדָמָּה

"עִצְרוּ, בָּנַי!" קָרָא קוֹל עָמֹק, רוֹעֵד וּמָּלֵא בִּקְדֻשָּׁה. הַיְּלָדִים הִסְתּוֹבְבוּ בְּבֶהָלָה. מוּלָם עָמַּד גְּדוֹל הַדּוֹר, רַבִּי יְהוּדָה בַּר אִלְעָאִי! מְעִילוֹ הַלָּבָן-אֲפֹר נָע בָּרוּחַ, וְעֵינָיו הַטּוֹבוֹת הִבִּיטִיו בָּהֶם בְּעֶצֶב כֹּה עָמֹק, עַד שֶׁכָּל הַצְּחוֹק שֶׁל הַיְּלָדִים קָפָא מִיָּד עַל שִׂפְתוֹתֵיהֶם.

הַיְּלָדִים הִשְׁפִּילוּ אֶת הַמַּבָּט בְּבוּשָׁה, אַךְ אָז רָאוּ מַחֲזֶה שֶׁזִּעְזַע אֶת נִשְׁמָּתָם: רַבִּי יְהוּדָה, הָאָדָם הֲכִי חָשׁוּב וּמְּכֻבָּד בַּגָּלִיל, לֹא צָעַק עֲלֵיהֶם וְלֹא גָּעַר בָּהֶם. בִּמְּקוֹמּוֹ זאת, הוּא כָּרַע אֶת בִּרְכָּיו הַזְּקֵנוֹת, וְהִתְכּוֹפֵף בַּעֲדִינוּת אֶל תּוֹךְ שְׁלוּלִית הַבּוֹץ הַמְּלֻכְלֶכֶת. בְּמוֹ יָדָיו הַטְּהוֹרוֹת וְהַקְּדוֹשׁוֹת, שֶׁבָּהֶן כָּתַב וְלִמֵּד מֵאוֹת הֲלָכוֹת, הוּא חָפַן אֶת הַלַּחְמַּנִיָּה הַמְעֻוֶּכֶת וְהַסְּפוּגָה בַּמַּיִם הַלּוּחֲצִים.

  • WE (אֲנַחְנוּ בַּיּוֹם-יוֹם): בַּדּוֹר שֶׁלָּנוּ, בָּרוּךְ ה', הַקָּדוֹשׁ בָּרוּךְ הוּא מַשְׁפִּיעַ עָלֵינוּ שֶׁפַע עָצוּם וְנִפְלָא. אֲנַחְנוּ פּוֹתְחִים אֶת הַמְּקָרֵר וְהוּא מָּלֵא בְּכָל טוּב, הָאִמָּא מְּכִינָה לָנוּ כְּרִיכִים מְּתוּקִים וּשְׁמֵנִים, וְלִפְעָמִּים – מֵּרֹב שֶׁיֵּשׁ לָנוּ הַכֹּל, אֲנַחְנוּ שוֹכְחִים לְהַעֲרִיךְ. אֲנַחְנוּ מַשְׁאִירִים כְּרִיכִים חֲצִי אֲכוּלִים בַּתִּיק, זוֹרְקִים לַחְמַּנִיּוֹת לַפַּח בְּלִי לַחְשֹׁב, אוֹ מִתְלוֹנְנִים: "לָמָּה הָאוֹכֶל הַזֶּה לא טָעִים לִי?".
  • YOU (אַתָּה, מָחָר בַּבֹּקֶר!): מָחָר בַּבֹּקֶר, כַּאֲשֶׁר תִּפְּתַּח אֶת תִּיק הָאוֹכֶל שֶׁלְּךָ בַּהַפְסָקָה וְתִרְאֶה אֶת הַכְּרִיךְ שֶׁאִמָּא הֵכִינָה לְךָ בְּאַהֲבָה וּבְמַּאֲמָּץ – עֲצֹר לִשְׁנִיָּה! זְכֹר אֶת רַבִּי יְהוּדָה וְאֶת הַלַּחְמַּנִיָּה שֶׁבַּבּוֹץ. בָּרֵךְ בְּכוֹחַ וּבְכַוָּנָה אֲמִּתִּית 'הַמּוֹצִיא', אֱכֹל בְּשִׂמְּחָה וּבְהַכָּרַת הַטּוֹב, וְתִזְכֶּה לִהְיוֹת צַדִּיק יָשָׁר וְקָדוֹשׁ הַמְּחֻבָּר לַבּוֹרֵא!

א. שְׁאֵלַת פְּתִיחָה

"יְלָדִים, אִם הֱיִיתֶם מּוֹצְאִים בָּרְחוֹב סְתָם פְּרוּסַת לֶחֶם יְבֵשָׁה אוֹ לַחְמַּנִיָּה זְרוּקָה בַּלִּכְלוּךְ – הַאִם הֱיִיתֶם מְּהַסְּסִים לִבְעֹט בָּהּ כְּמוֹ בְּכַדּוּר, אוֹ שֶׁמָּא הֱיִיתֶם מְבִינִים שֶׁיֵּשׁ בָּהּ קְדֻשָּׁה וְעָמַּל עָצוּם? הַיּוֹם נִשְׁמַּע עַל גְּדוֹל הַדּוֹר, רַבִּי יְהוּדָה בַּר אִלְעָאִי, שֶׁהִפְתִּיעַ אֶת כָּל הַיְּלָדִים בַּשּׁוּק בִּגְלַל דָּבָר שֶׁמָּצָא בַּבּוֹץ!".

ב. הַעֲרוֹת בִּצּוּעַ וּסְטוֹרִיטֶלִינְגּ

  • שְׁקוּפִית 8-9: דַּבְּרוּ בִּקְצָת יוֹתֵר כֹּחַ וּבְשִׂמְּחָה כַּאֲשֶׁר אַתֶּם מְּפָרְטִים אֶת עֲשֶׂרֶת עֲמָּלֵי הַלֶּחֶם, וַעֲשׂוּ תְּנוּעוֹת שֶׁל חֲרִישָׁה, זְרִיעָה וַאֲפִיָּה בַּיָּדַיִם כְּדֵי לִסְחוֹף אֶת הַבָּנִים.

ג. הַמְּלָצוֹת לְמִּצְגַּת (4 שְׁקוּפִיּוֹת קְדוֹשׁוֹת וּצְנוּעוֹת)

  • שְׁקוּפִית 1 — כּוֹתֶרֶת: הַלַּחְמַּנִיָּה שֶׁבַּבּוֹץ. אִיּוּר מְרַגֵּשׁ וּצְנוּעַ שֶׁל רַבִּי יְהוּדָה בַּר אִלְעָאִי (עִם זָקָן לָבָן מְּפֹאָר וְכִיפָּה גְּדוֹלָה על רֹאשׁוֹ) מַבִּיט בַּחִיּוּךְ מֵאִיר, כְּשֶׁבָּרֶקַע קַרְנֵי אוֹר שְׁמֵּיְמִּיּוֹת מְּכַסּוֹת אֶת בֵּית הַמִּדְרָשׁ הָעַתִּיק.
  • שְׁקוּפִית 2 — הַמִּשְׂחָק בַּסִּמְּטָה. אִיּוּר שֶׁל בָּנִים חֲמוּדִים בַּגָּלִיל (לְבוּשִׁים צָנוּעַ בְּשַׁרְווּלִים אֲרֻכִּים וְכִיפּוֹת עַל רֹאשָׁם) מְּשַׂחֲקִים בַּשּׁוּק בְּעֶרֶב חֹרֶף. הַלַּחְמַּנִיָּה הַזְּהֻבָּה נִרְאֵית עָפָה בָּאֲוִיר, כְּשֶׁלְּמַטָּה שְׁלוּלִית בּוֹץ לַחָה.
  • שְׁקוּפִית 3 — הַדְּמָּעוֹת שֶׁל גְּדוֹל הַדּוֹר (הַשִּׂיא). אִיּוּר מָלֵא קְדֻשָּׁה וְהוֹד. רַבִּי יְהוּדָה הַצַּדִּיק כּוֹרֵעַ בַּעֲנָוָה, אָסַף אֶת הַלַּחְמַּנִיָּה הַמְלֻכְלֶכֶת בְּיָדָיו הַמְּבֻצָּקוֹת. דְּמָּעָה נוֹצֶצֶת נִשְׁקֶפֶת מֵּעֵינָיו אֶל מוּל הַלֶּחֶם, וְהַיְּלָדִים עָמְּדוּ מִּסָּבִיב בְּתַחְדֵּשׁ וּבִתְשׁוּבָה.
  • שְׁקוּפִית 4 — שִׂמְּחַת הַמִּצְוָה וְהַבְּרָכָה. אִיּוּר מֵאִיר וּמְּקָרֵב. רַבִּי יְהוּדָה עוֹמֵּד וּמְּבָרֵךְ אֶת הַבָּנִים הַחֲמוּדִים, הַמַּבִּיטִים בַּשָּׁמַּיִם בְּאַהֲבָה וּבֶאֱמוּנָה. בְּיָדָיו שֶׁל נְתַנְאֵל הַיֶּלֶד יֵשׁ כְּרִיךְ נָקִי וּמְּעֻטָּר, וְעָלָיו כְּתוּבָה בְּאוֹתִיּוֹת זָהָב הַבְּרָכָה 'הַמּוֹצִיא'.

ד. פְּעִילוּת מְסַכֶּמֶּת: "הַלַּחְמַּנִיָּה שֶׁל הַכַּוָּנָה"

הַשֶּׁמֶּשׁ הַזְּהֻבָּה שֶׁל אֶרֶץ יִשְׂרָאֵל שָׁלְחָה קַרְנֵי אוֹר רַכּוֹת וַחֲמִימוֹת מֵעַל חוֹמוֹת הָאֶבֶן הַלְּבָנוֹת וְהַמַּלְכוּתִיּוֹת שֶׁל יְרוּשָׁלַיִם. בָּאֲוִיר הַזַּךְ וְהַמְּעוֹרֵר נֶפֶשׁ, עָמַּד רֵיחַ מָתוֹק וְקָדוֹשׁ שֶׁל קְטֹרֶת הַבְּשָׂמִים הָעוֹלֶה מִבֵּית הַמִּקְדָּשׁ הַשֵּׁנִי. הַסִּמְּטָאוֹת הָיוּ הומּוֹת בְּקוֹלוֹת שִׂמְּחָה, בְּשִׁירַת כִּנּוֹרוֹת וּבְנִגּוּנִים נִפְלָאִים שֶׁל עוֹלֵי הָרֶגֶל שֶׁנָּהֲרוּ מִכָּל קְצוֹת הָאָרֶץ לַחְגֹּג אֶת אהֲבַת ה'. יְרוּשָׁלַיִם הָיְתָה יְפֵהפִיָּה, כְּלִילַת יֹפִי, וְהַבָּתִּים הַמְּפֹאָרִים שֶׁל עֲשִׁירֵי הָעִיר עָמְדוּ בִּתְהִלָּתָם.

בְּתוֹךְ אֶחָד מִן הָאַרְמוֹנוֹת הַגְּדוֹלִים, הַמְּעֻטָּרִים בְּעַמּוּדֵי שַׁיִשׁ וּבִמְּנוֹרוֹת זָהָב נוֹצְצוֹת, גָּר אָדָם עָשִׁיר מְאֹד. הָעָשִׁיר הַזֶּה הָיָה מְכֻבָּד בָּעִיר, אַךְ בְּתוֹךְ לִבּוֹ שָׁכְנָה מִדָּה קָשָׁה שֶׁל גַּאֲוָה וְשִׂנְאָה. הָיוּ לוֹ שְׁנֵי אֲנָשִׁים שֶׁהִכִּיר: אָדָם אֶחָד שֶׁקָּרְאוּ לוֹ קַמְצָא, שֶׁהָיָה חֲבֵרוֹ הַטּוֹב וְהָאָהוּב עָלָיו מִכָּל; וְאָדָם שֵׁנִי שֶׁקָּרְאוּ לוֹ בַּר קַמְצָא, שֶׁהָיָה שׂוֹנְאוֹ הַמֻּבְהָק וּבַר-פְּלֻגְתָּא שֶׁלּוֹ.

לִכְבוֹד אֵרוּעַ מְשַׂמֵּחַ בְּמִשְׁפַּחְתּוֹ, הֶחְלִיט הָעָשִׁיר לַעֲרֹךְ סְעוּדַת מְלָכִים עֲצוּמָה וּמְפֹאָרֶת. הַשֻּׁלְחָנוֹת נֶעֶרְכוּ בְּמַפּוֹת מֶשִׁי צְחוֹרוֹת, וְהַמִּטְבָּח הִתְמַּלֵּא בְּרֵיחַ נִפְלָא וּמְּעוֹרֵר תֵּאָבוֹן שֶׁל בָּשָׂר צָלוּי, חַלּוֹת חַמּוֹת וִיקָרוֹת וְיַיִן מָתוֹק וְעַתִּיק. הָעָשִׁיר קָרָא לִשְׁלִיחוֹ הַנֶּאֱמָּן וְאָמַּר לוֹ: "בְּנִי, קַח אֶת כַּרְטִיס הַהַזְמָּנָה הַזּוֹ, רוּץ מַהֵר בַּסִּמְּטָאוֹת, וּמְּצָא אֶת אוֹהֲבִי קַמְצָא – הָבֵא אוֹתוֹ אֵלַי כְּדֵי שֶׁיִּשְׂמַּח עִמִּי בִּסְעוּדָתִי!".

הַטָּעוּת הַנּוֹרָאָה וְהַהַזְמָּנָה הַמֻּטְעֵית

הַשָּׁלִיחַ, שֶׁהָיָה נָבוֹךְ וּמְּמַּהֵר מְּאֹד מִפְּנֵי כְּנִיסַת הַחַג, רָץ בִּזְרִיזוּת וְלֹא שָׂם לֵב לַשֵּׁמוֹת הַדּוֹמִּים. בִּמְּקוֹם לָלֶכֶת אֶל הָאוֹהֵב קַמְצָא, הָלַךְ בְּטוּת וּפָסַע אֶל בֵּיתוֹ שֶׁל הַשּׂוֹנֵא – בַּר קַמְצָא.

טּוּק... טּוּק... טּוּק... נִשְׁמַּע עַל דֶּלֶת הַבַּיִת שֶׁל בַּר קַמְצָא. בַּר קַמְצָא פָּתַח אֶת הַדֶּלֶת. מוּלוֹ עָמַּד הַשָּׁלִיחַ הַמַּבְהִיל: "שָׁלוֹם לְךָ, הָעָשִׁיר הַגָּדוֹל מַזְמִּין אוֹתְךָ הָעֶרֶב לִסְעוּדַת הַמִּשְׁתֶּה הַמְפֹאָרָה שֶׁלּוֹ!". בַּר קַמְצָא נִדְהַם. הַבֶּטֶן שֶׁלּוֹ הִתְהַפְּכָה מֵרֹב הִתְרַגְּשׁוּת וְתִקְוָה. "הַאִם יִתָּכֵן?" חָשַׁב בַּר קַמְצָא בְּדִמְּעָה שֶׁל שִׂמְּחָה בְּעֵינָיו, "הַאִם שׂוֹנְאִי הַגָּדוֹל הֶחְלִיט לַעֲשׂוֹת עִמָּדִי שָׁלוֹם? הַאִם הַקָּדוֹשׁ בָּרוּךְ הוּא פָּתַח אֶת לִבּוֹ לִנְהֹג בְּאַהֲבַת חִנָּם וּלְחַבֵּק אוֹתִי מִחָדָשׁ?". מִתּוֹךְ רָצוֹן עָז לְהַשְׁלִים, לָבַשׁ בַּר קַמְצָא אֶת בִּגְדֵי הַחַג הַיָּפִים שֶׁלּוֹ, עָטַף אֶת עַצְמּוֹ בִּגְלִימָה מְכֻבֶּדֶת, שָׂם כִּיפָּה יָפָה עַל רֹאשׁוֹ וּמִּהֵר בִּצְעָדִים קַלִּים אֶל אֲוִירַת הַחַג שֶׁבָּאַרְמוֹן.

כַּאֲשֶׁר נִכְנַס פְּנִימָּה, הָעוֹלָם נִרְאָה כְּחָלוֹם. הָאוּלָם הָיָה הוֹמֶּה וּמָּלֵא מִקָּצֶה אֶל קָצֶה. מֵאוֹת אֲנָשִׁים, וּבָהֶם כָּל גְּדוֹלֵי וַחֲכָמֵּי יִשְׂרָאֵל, יָשְׁבוּ מְחַיְּכִים, שׁוֹתִים יַיִן וְאוֹכְלִים מַעֲדַנִּים קְדוֹשִׁים. בַּר קַמְצָא הִתְיַשֵּׁב בִּקְצֵה הַשֻּׁלְחָן בְּלֵב רוֹעֵד מֵאַהֲבָה וּבִטָּחוֹן בַּשָּׁלוֹם.

הַתַּפְנִית הָאַכְזָרִית וְזַעֲמוֹ שֶׁל בַּעַל הַבַּיִת

לְפֶתַע, נִכְנַס בַּעַל הַבַּיִת הָעָשִׁיר אֶל הָאוּלָם. פְּנֵי הָעָשִׁיר הָיוּ מְחַיְּכוֹת לְכֻלָּם, אַךְ כַּאֲשֶׁר הֵנִיף אֶת מַבָּטוֹ וְסָרַק אֶת הַסְּעוּדָה, עֵינָיו נִתְקְעוּ בִּדְמּוּתוֹ שֶׁל בַּר קַמְצָא הַיּוֹשֵׁב בַּקָּצֶה. פִּתְאוֹם פָּנָיו שֶׁל הָעָשִׁיר אָדְמוּ מִכַּעַס נורָא. חִיּוּכוֹ נֶעֱלַם וְהוּא פָּסַע בִּצְעָדִים כְּבֵדִים, מַרְעִישִׁים וּמְּאַיְּמִּים יְשִׁירוֹת אֶל עֶבְרוֹ שֶׁל בַּר קַמְצָא.

"מָה אַתָּה עוֹשֶׂה כָּאן?!" שָׁאַג הָעָשִׁיר בְּקוֹל שֶׁהִרְעִישׁ אֶת כָּל הַחֶדֶר וְגָרַם לְכָל הַכִּנּוֹרוֹת לְהִשְׁתַּתֵּק מִיָּד. "הֲרֵי שׂוֹנֵא לִי אַתָּה! קוּם מִיָּד וְצֵא מִבֵּיתִי! אֵינִי רוֹצֶה שֶׁתַּעֲמֹד כָּאן אֲפִלּוּ לְשָׁנִיָּה יְחִידָה!".

בַּר קַמְצָא הִרְגִּישׁ כְּאִלּוּ מַיִם קָרִים וּקְפוּאִים נִשְׁפְּכוּ עַל גּוּפוֹ. כָּל מַבָּטֵי הָאוֹרְחִים הֻפְנוּ אֵלָיו בִּשְׁתִיקָה מְמַהֶרֶת. פָּנָיו שֶׁל בַּר קַמְצָא הָפְכוּ אֲדֻמּוֹת כְּאֵשׁ מִּבּוּשָׁה. הַחֲכָמִּים הַצַּדִּיקִים הָיוּ שָׁם, יָשְׁבוּ עַל סַפְסְלֵיהֶם, אַךְ... אַף אֶחָד לא קָם לַעֲצֹר אֶת הָעֶלְבּוֹן. אַף קוֹל לא נִשְׁמַּע לְהָגֵן על כְּבוֹדוֹ שֶׁל הָאָדָם הַמֻּסְכֵּן.

הַדִּילֶמָּה וְהַבַּקָּשׁוֹת הַנּוֹאָשׁוֹת לִמְנִיעַת הַבּוּשָׁה

בַּר קַמְצָא הִבִּיט בַּחֲכָמִּים בְּתִקְוָה, מְּבַקֵּשׁ בְּעֵינָיו שֶׁמִּיהוּ יָקוּם וְיֹאמַּר מִלָּה. אַךְ הַשֶּׁקֶט הָיָה כָּבֵד וְנוֹרָא. הוּא פָּנָה אֶל בַּעַל הַבַּיִת בִּלְחִישָׁה רוֹעֶדֶת, אוֹחֵז בִּקְצֵה גְּלִימָתוֹ בְּהַכְנָעָה: "אָנָּא, אֲדוֹנִי! מִּכֵּיוָן שֶׁכְּבָר הִגַּעְתִּי וַאֲנִי יוֹשֵׁב כָּאן, אַל תְּבַקֵּשׁ לְבַיֵּשׁ אוֹתִי בָּרַבִּים. הַנַּח לִי לֵישֵׁב וֶאֱכֹל, וַאֲנִי מִּתְחַיֵּב לְשַׁלֵּם לְךָ מִכַּסְפִּי הָאִישִׁי עֲבוּר כָּל מָה שֶׁאֹכַל וְאֶשְׁתֶּה הָעֶרֶב!".

"לא!" צָעַק הָעָשִׁיר, קַשְׁיוּתוֹ כָּאֶרֶז הָיְתָה קָשָׁה מִבַּרְזֶל. "קוּם וְצֵא מִיָּד!".

בַּר קַמְצָא הָיָה נוֹאָשׁ. הַדְּמָּעוֹת הִתְחִילוּ לַחְנֹק אֶת גְּרוֹנוֹ. "אֲדוֹנִי!" הִתְחַנֵּן שֵׁנִית, "מוּכָן אֲנִי לְשַׁלֵּם חֲצִי מִכָּל הַהוֹצָאוֹת הָעֲצוּמוֹת שֶׁל הַסְּעוּדָה כֻּלָּהּ! רַק אַל תַּשְׁלִיךְ אוֹתִי מִפֹּה בְּבוּשָׁה כָּזוֹ מוּל כָּל גְּדוֹלֵי יִשְׂרָאֵל!".

"לֹא!" זָעַק בַּעַל הַבַּיִת, פָּנָיו מְּעֻוָּתוֹת מִשִּׂנְאָה.

בַּר קַמְצָא רָעַד. בִּמְסִירַת נֶפֶשׁ אַחְרוֹנָה, מוּכָן לְוַתֵּר עַל כָּל הוֹנוֹ, קָרָא: "אֲדוֹנִי! אֲנִי אֲשַׁלֵּם לְךָ אֶת כָּל הַהוֹצָאוֹת שֶׁל הַסְּעוּדָה וְהַמִּשְׁתֶּה כֻּלּוֹ! כָּל מַטְבְּעוֹת הַזָּהָב וְהַכֶּסֶף שֶׁלִּי אֶתֵּן לְךָ, וּבִלְבַד שֶׁלֹּא אֵצֵא בְּבוּשָׁה נוֹרָאָה שֶׁכָּזוֹ!".

שורה X: "כָּאן – עֲצֹר. הַבֵּט עַל הַיְּלָדִים בְּמַּבָּט נִרְעָשׁ. שְׁאַל בְּקוֹל רוֹעֵד: 'יְלָדִים, אִם אָדָם מוּכָן לְשַׁלֵּם סְכוּם כָּל כָּךְ עָצוּם, הַכֹּל כְּדֵי שֶׁלֹּא יַלְבִּינוּ אֶת פָּנָיו בָּרַבִּים – מָּה לְדַעְתְּכֶם יַעֲשֶׂה הָעָשִׁיר הַגֵּאֶה? הַאִם לִבּוֹ יִתְרַכֵּךְ, אוֹ שֶׁמָּא הַשִּׂנְאָה תְּנַצֵּחַ?'"

הַשִּׂיא הַמַּכְאִיב – הַגֵּרוּשׁ הַנּוֹרָא וּשְׁתִיקַת הַחֲכָמִּים

הָעָשִׁיר לא רָצָה כֶּסֶף. הַשִּׂנְאָה שֶׁבְּתוֹךְ לִבּוֹ הָיְתָה כֹּה עֲמוּקָה, עַד שֶׁסִּמְּתָה אֶת עֵינָיו. הוּא צָעַד קָדִימָּה, תָּפַס בְּכֹחַ וּבְאַכְזָרִיּוּת בִּבְגָדוֹ הַיָּפֶה שֶׁל בַּר קַמְצָא, הֵקִים אוֹתוֹ מִּכִּסֵּאוֹ, וּבְעֵינַיִם מְלֵאוֹת כַּעַס גֵּרֵשׁ וְהִשְׁלִיךְ אוֹתוֹ הַחוּצָה בִּבְעִיטָה, נוֹעֵל אֶת הַדֶּלֶת הַכְּבֵדָה בִּנְעִילָה אַחֲרוֹנָה וּבִרְעָשׁ גָּדוֹל: טְרָאח!.

בַּר קַמְצָא עָמַּד בָּרְחוֹב הַחָשׁוּךְ. דִּמְעוֹתָיו הַחַמּוֹת וְהַזַּכּוֹת זָלְגוּ עַל הֶעָפָר. הַקֹּר שֶׁל הַלַּיְלָה צָבַט אֶת בְּשָׂרוֹ, אַךְ הַכְּאֵב שֶׁבְּתוֹךְ לִבּוֹ הָיָה קָשֶׁה מִּקֹּר הַשֶּׁלֶג. "הַחֲכָמִּים יָשְׁבוּ שָׁם," לָחַשׁ בַּר קַמְצָא בְּכֶאֶב שׁוֹרֵף. "גְּדוֹלֵי הַדּוֹר רָאוּ אֶת עֶלְבּוֹנִי הַנּוֹרָא וְשָׁתְקוּ. שְׁמַּע מִּינָּה שֶׁהֵם מַּסְכִּמִּים לַבִּיּוּשׁ שֶׁלִּי! שְׁמַּע מִּינָּה שֶׁנַּפְשׁוֹ שֶׁל אָדָם פָּשׁוּט אֵינֶנָּה חֲשׁוּבָה בְּעֵינֵיהֶם!".

מִתּוֹךְ כְּאֵב הָעֶלְבּוֹן וְהַשִּׂנְאָה שֶׁנִּבְרְאָה בּוֹ עַל שְׁתִיקַת הַסּוֹבֲבִים, הָלַךְ בַּר קַמְצָא וְעָשָׂה מַעֲשֶׂה אָיֹם. הוּא הָלַךְ אֶל הַקֵּיסָר הָרוֹמָּאִי וְאָמַּר לוֹ: "הַיְּהוּדִים מּוֹרְדִים בְּךָ!". כְּדֵי לְהוֹכִיחַ זאת, שָׁלַח הַקֵּיסָר עֵגֶל קָדוֹשׁ לְקָרְבָּן בְּבֵית הַמִּקְדָּשׁ. בַּר קַמְצָא, בְּדַרְכּוֹ, עָשָׂה כֶּתֶם קָטָן בִּשְׂפָתוֹ שֶׁל הָעֵגֶל (דָּבָר שֶׁפּוֹסֵל אֶת הַקָּרְבָּן לְפִי דִּינֵי הַתּוֹרָה, אַךְ אֵינוֹ פּוֹסֵל אֶצְל הָרוֹמָּאִים).

כַּאֲשֶׁר סֵרְבוּ הַכֹּהֲנִים לְהַקְרִיב אֶת הַקָּרְבָּן הַפָּסוּל בִּשְׁבִיל שְׁמִּירַת כְּבוֹד הַתּוֹרָה וְהַמִּזְבֵּחַ, כָּעַס הַקֵּיסָר הָרוֹמָּאִי מְּאֹד. הוּא שָׁלַח צָבָא עָצוּם, שֶׁעָלָה עַל יְרוּשָׁלַיִם, וּבִגְלַל שִׂנְאַת חִנָּם וְהַלְבָּנַת פָּנִים – נֶחְרַב בֵּית הַמִּקְדָּשׁ הַשֵּׁנִי וְעַם יִשְׂרָאֵל יָצָא לַגָּלוּת הָאֲרֻכָּה.

שורה Y: "כָּאן – עֲצֹר. דַּבֵּר לְאַט, בְּקוֹל שָׁקֵט וּמָלֵא בִּדְמָּמָּה רוֹעֶדֶת: 'הַנֵּר שֶׁל הַמִּקְדָּשׁ כָּבָה... אַךְ הַקָּדוֹשׁ בָּרוּךְ הוּא הִבְטִיחַ לָנוּ שֶׁאִם נֵדַע לִנְהֹג בְּאַהֲבַת חִנָּם וּבִרְגִישׁוּת, נִזְכֶּה לִבְנוֹת אֶת בֵּית הַמִּקְדָּשׁ מֵחָדָשׁ!'".

💎 חִבּוּר עֶרְכִּי: ME - WE - YOU

  • ME (בַּר קַמְצָא וְהַחֲכָמִּים): בַּר קַמְצָא רָצָה מְאֹד לְהַשְׁלִים, אַךְ הַשִּׂנְאָה שֶׁל הָעָשִׁיר וְהַשְּׁתִיקָה שֶׁל הַסּוֹבֲבִים פָּגְעוּ בּוֹ בְּאֹפֶן כֹּה קָשֶׁה. מִתּוֹךְ כָּךְ נִלְמַד שֶׁשְּׁתִיקָה מוּל עֶלְבּוֹן שֶׁל חָבֵר הִיא שֻׁתָּפוּת בַּחֵטְא מַמָּשׁ.
  • WE (אֲנַחְנוּ בַּיּוֹם-יוֹם): גַּם אֶצְלָנוּ, לִפְעָמִּים בַּכִּתָּה אוֹ בַּמִּשְׂחָק, הַיֵּצֶר הָרַע מְּפַתֶּה אוֹתָנוּ לִלְעֹג לְחָבֵר, לֹא לְשַׁתֵּף יֶלֶד בַּמִּשְׂחָק, אוֹ – וְזֶה הֲכִי נָפוֹץ – לִעֲמֹד בַּצַּד וְלִשְׁתֹּק כְּשֶׁיֶּלֶד אַחֵר מְּבַיֵּשׁ חָבֵר בַּכִּתָּה, רַק כְּדֵי "שֶׁלֹּא יִצְחֲקוּ גַּם עָלַי".
  • YOU (אַתָּה, מָחָר בַּבֹּקֶר!): מָחָר בַּבֹּקֶר, כַּאֲשֶׁר תֵּלֵךְ לְבֵית הַסֵּפֶר וְתִרְאֶה חָבֵר שֶׁעוֹמֵד בַּצַּד, שֶׁמִּתְקַשֶּׁה בַּמִּשְׂחָק אוֹ שֶׁמִּיהוּ מְּדַבֵּר אֵלָיו בְּלא כָּבוֹד – אל תִּשְׁתֹּק! עֲצֹר, גַּשׁ אֵלָיו, חַיֵּךְ אֵלָיו, וֶאֱמֹּר לוֹ מִּלָּה טוֹבָה. בַּמַּעֲשֶׂה הַקָּטָן הַזֶּה שֶׁל אַהֲבַת חִנָּם, אַתָּה מְּכַבֶּה אֶת הָאֵשׁ שֶׁחֵרְבָה אֶת הַבַּיִת, וּמֵּבִיא עָלֵינוּ אֶת הַבְּרָכָה וְהַשָּׁלוֹם לְתָמִיד.

👨‍🏫 מַדְרִיךְ לַמּוֹרֶה / לַמְּסַפֵּר

א. שְׁאֵלַת פְּתִיחָה

"יְלָדִים, אִם הֱיִיתֶם רוֹאִים חָבֵר שֶׁמְּבַיֵּשׁ יֶלֶד אַחֵר מוּל כָּל הַכִּתָּה, מָה הֱיִיתֶם עוֹשִׂים? הַאִם הֱיִיתֶם מִּתְעָרְבִים וּמַּצִּילִים אוֹתוֹ, אוֹ שֶׁמָּא הֱיִיתֶם מְּפַחֲדִים וְשׁוֹתְקִים? הַיּוֹם נִשְׁמַּע עַל יוֹם אֶחָד וְעָצוּב בִּירוּשָׁלַיִם, שֶׁבּוֹ שְׁתִיקָה שֶׁל אֲנָשִׁים מוּל עֶלְבּוֹן – הֵבִיאָה לַחֻרְבָּן הֲכִי כָּבֵד בַּהִיסְטוֹרְיָה!".

ב. הַעֲרוֹת בִּצּוּעַ וּסְטוֹרִיטֶלִינְג

  • שׁוּרָה 18 (הַטָּעוּת): "דַּבֵּר מַהֵר, בְּקוֹל מּוֹתֵחַ וְעַלִּיז כְּדֵי לְהַעֲצִים אֶת הַמֶּתַח שֶׁל הַטָּעוּת".
  • שׁוּרָה 33 (הַדִּילֶמָּה): "כָּאן – עֲצֹר. פְּרֹשׂ אֶת יָדֶיךָ, הַבֵּט בַּיְּלָדִים בְּעֵינַיִם רַחֲבֹות וּבְעֶצֶב שָׁקֵט".
  • שׁוּרָה Y (הַשְּׁתִיקָה): "הַנְמִּיךְ אֶת הַקּוֹל לִלְחִישָׁה מְּרַגֶּשֶׁת. זֶהוּ רֶגַע הַהַכְנָעָה וְהַבּוּשָׁה הַמַּכְאִיבָה".

ג. הַמְלָצוֹת לַמַּצֶּגֶת (צְנִיעוּת וּקְדֻשָּׁה)

  • שְׁקוּפִית 1: הַשּׁוּק הַהוֹמֶּה וְהַזּוֹהֵר שֶׁל יְרוּשָׁלַיִם, בָּרֶקַע מִתְרוֹมֵם בֵּית הַמִּקְדָּשׁ הַשֵּׁנִי בְּתִפְאַרְתּוֹ, אוֹר שְׁמֵּיְמִי וְזָהָב. הָאֲנָשִׁים הַלּוֹבְשִׁים שַׁרְווּלִים אֲרֻכִּים וְעוֹטִים כִּיפּוֹת.
  • שְׁקוּפִית 2: בַּר קַמְצָא פּוֹתֵחַ אֶת הַדֶּלֶת לַשָּׁלִיחַ, פְּנֵי בַּר קַמְצָא מְאִירוֹת בְּתִקְוָה, כִּיפָּה בְּרוּרָה וּלְבָנָה עַל רֹאשׁוֹ.
  • שְׁקוּפִית 3: אָלּוֹן הַמִּשְׁתֶּה הַמְּפֹאָר בָּאַרְמוֹן. הָעָשִׁיר אוֹחֵז בְּיָדוֹ שֶׁל בַּר קַמְצָא, וּבַר קַמְצָא מַשְׁפִּיל אֶת עֵינָיו בְּבוּשָׁה מְלֵאִים בִּדְמָּעוֹת. הַחֲכָמִּים הַמְּעֻטָּפִים בְּטַלִּיתוֹת וְשַׁרְווּלִים יושְׁבִים מִסָּבִיב בְּשֶׁקֶט.
  • שְׁקוּפִית 4: בַּר קַמְצָא עוֹמֵּד בַּחוּץ בַּחֹשֶׁךְ מֵאֲחוֹרֵי הַדֶּלֶת הַנְּעוּלָה, מַבִּיט לַשָּׁמַּיִם, וְדִמְּעָה זוֹהֶרֶת נוֹפֶלֶת מֵעֵינָיו אֶל אֲדָמַּת הַקֹּדֶשׁ שֶׁל יְרוּשָׁלַיִם.

ד. פְּעִילוּת מְּסַכֶּמֶּת

  • מִּשְׂחַק תַּפְקִידִים שֶׁל אַהֲבַת חִנָּם (מְתַקְּנִים אֶת הַסִּפּוּר!):

"הַבָּנִים מִּתְחַלְּקִים לִשְׁלֹשָׁה: אֶחָד הָעָשִׁיר, הַשֵּׁנִי בַּר קַמְצָא הַנִּכְנָס, וְהַשְּׁלִישִׁי הוּא תַּלְמִּיד חָכָם מֵהַכִּתָּה. כַּאֲשֶׁר הָעָשִׁיר מַתְחִיל לִצְעֹק, הַתַּלְמִּיד מִתְעָרֵב מִיָּד בְּאַהֲבָה וּבַחִיּוּךְ, מְקָרֵב אֶת הַכִּסֵּא עֲבוּר בַּר קַמְצָא וְאוֹמֵּר: 'בּוֹא, שֵׁב לְיָדִי, כֻּלָּנוּ בָּנִים שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא וּמִּצְוָה עָלֵינוּ לִשְׂמֹּחַ יַחַד!'".

ב. הַעֲרוֹת בִּצּוּעַ וּסְטוֹרִילַּנְג (הַנְחָיָה קוֹלִית וּפִיזִית)

ג. הַמְלָצוֹת לַמַּצֶּגֶת (תֵּאוּר 4 שְׁקוּפִיּוֹת בִּקְדֻשָּׁה וּצְנִיעוּת)

  • שְׁקוּפִית 2 – הַמִּפְגָּשׁ בַּמִּדְבָּר: אָבָּא תַּחְנָא כּוֹרֵעַ בְּעָנָוָה לִפְנֵי הַחוֹלֶה (הַמְּכֻסֶּה בִּבְגָדִים אֲרֻכִּים וּצְנוּעִים), מַּבִּיט בּוֹ בְּחֶמְּלָה וּבְיוֹשֶׁר, חֲבִילַת הָעֵצִים מֻּנַּחַת בַּצַּד.
  • שְׁקוּפִית 3 – הַשֶּׁמֶּשׁ שֶׁעָצְרָה: תְּמּוּנַת שַׁעַר הָעִיר, הַשֶּׁמֶּשׁ הַזּוֹהֶרֶת פִּתְאוֹם בְּאוֹר אֱלֹקִי מְּתוּק וְחָם מִּבַּעַד לָעֲנָנִים, מַּרְטִיבָה אֶת הַכֹּל בְּבָרָק זָהָב. אָבָּא תַּחְנָא וְאַנְשֵׁי הָעִיר (עִם כִּיפּוֹת) מַּבִּיטִים לַשָּׁמַּיִם בְּעֵינַיִם פְּתוּחוֹת שֶׁל הוֹדָיָה.
  • שְׁקוּפִית 4 – הַבַּת קוֹל וְהַשִּׂמְּחָה: אָבָּא תַּחְנָא יוֹשֵׁב בְּשִׂמְּחָה עִם מִּשְׁפַּחְתּוֹ מִּסָּבִיב לְשֻׁלְחַן הַשַּׁבָּת הַמֵּאִיר, נֵרוֹת שַׁבָּת דּוֹלְקִים בְּרוֹךְ, וְאוֹר שְׁמֵּימִּי נִפְלָא יוֹרֵד מִּלְּמַּעְלָה, מְּסַמֵּל אֶת דִּבְרֵי הַבַּת קוֹל.

ד. פְּעִילוּת מְּסַכֶּמֶּת

  • מִּשְׂחַק תַּפְקִידִים – "פָּרָשַׁת הַדְּרָכִים שֶׁלִּי": הַמּוֹרֶה מֵּסִיב אֶת הַבָּנִים בַּכִּתָּה לִשְׁלָשׁוֹת. אֶחָד מְּשַׂחֵק אֶת אָבָּא תַּחְנָא, הַשֵּׁנִי אֶת יֵצֶר הָרַע הַמַּפְחִיד אוֹתוֹ עַל הַמָּמּוֹן, וְהַשְּׁלִישִׁי אֶת יֵצֶר הַטּוֹב הַמַּזְכִּיר לוֹ אֶת חֲשִׁיבוּת הַחֶסֶד וְהַשָּׁלוֹם. הַבָּנִים מְּתַרְגְּלִים כֵּיצַד יֵצֶר הַטּוֹב "מְּנַצֵּחַ" בַּדּוּ-שִׂיחַ, כְּדֵי לְהַפְנִים אֶת הַכְּלָל שֶׁלָּמַּדְנוּ!

🔹 בֵּרוּר מַקְדִּים (הֲכָנָה לַמּוֹרֶה/לַמְּסַפֵּר)

  • דְּמוּת הַגִּבּוֹר: הַתַּנָּא וְהָאָמוֹרָא הַקָּדוֹשׁ רַבִּי יַנַּאי, מִגְּדוֹלֵי חַכְמֵּי אֶרֶץ יִשְׂרָאֵל בַּגָּלִיל, אֲשֶׁר הֵקִים יְשִׁיבָה מְפֹאָרֶת בָּעִיר עַכְבָּרָא, וְהָיָה יָדוּעַ בַּחֲסִידוּתוֹ וּבְעַנְוָתוֹ הַמֻּפְלָאָה. לְצִדּוֹ עָמַּד הָרוֹכֵל הַפָּשׁוּט – סוֹחֵר נַוָּד הַסּוֹבֵב בֵּין הַכְּפָרִים וְהַשְּׁוָקִים כְּדֵי לִמְכֹּר אֶת מַּרְכּוּלְתּוֹ.
  • הַקּוֹנְפְלִיקְט / הַבְּעָיָה: רַבִּי יַנַּאי, שֶׁהִקְדִּישׁ אֶת כָּל חַיָּיו לְלִמּוּד תּוֹרָה בְּעַמְּקוּת, הִרְגִּישׁ שֶׁפְּסוּקֵי סֵפֶר תְּהִלִּים נִשְׁאָרִים בְּעֵינָיו לִפְעָמִּים כְּמֻּשָׁגִים עֶלְיוֹנִים וּמֻּפְשָׁטִים, וְלֹא כְּדֶּרֶךְ חַיִּים פְּשׁוּטָה הַמִּתְבַּטֵּאת בְּמַּעֲשֵׂי הַיּוֹם-יוֹם הַפְּשׁוּטִים שֶׁל הַהֲמוֹנִים בַּשּׁוּק.
  • נְקֻדַּת הַשִּׂיא: הָרוֹכֵל עוֹלֶה אֶל חַדְרוֹ שֶׁל הָרב, פּוֹתֵחַ סֵפֶר תְּהִלִּים קָטָן וּמַּצְבִּיעַ בְּאֶצְבַּע מְּאֻבֶּקֶת וּמְּחֻסְפָּסָה עַל הַפָּסוּק הַמֻּכָּר: "מִּי הָאיִשׁ הֶחָפֵץ חַיִּים... נְצֹר לְשׁוֹנְךָ מֵּרָע!".
  • מוּסַר הַשְּׂכֵל: "סַם הַחַיִּים" הָאֲמִּתִּי וְהַדֶּרֶךְ לִזְכּוֹת בְּחַיִּים מְּאֻשָּׁרִים וּבְרוּכִים מֵהַקָּדוֹשׁ בָּרוּךְ הוּא אֵינָם דּוֹרְשִׁים תְּרוּפוֹת פֶּלֶא גַּשְׁמִּיּוֹת בְּבַקְבּוּקִים, אֶלָּא אֶת הַשְּׁמִּירָה הַפְּשׁוּטָה וְהַקְּדוֹשָׁה עַל הַפֶּה וְהַלָּשׁוֹן מִּלְּדַבֵּר לָשׁוֹן הָרָע, רְכִילוּת אוֹ מִלִּים הַמַּעֲלִיבוֹת וּפוֹגְעוֹת בַּחֲבֵרִים.

📖 הַסִּפּוּר הַמָּלֵא: הָרוֹכֵל הַפָּשׁוּט וְסַם הַחַיִּים הַקָּדוֹשׁ

א. הַקּוֹל הַמִּסְתּוֹרִי בַּשּׁוּק שֶׁל הַגָּלִיל

הַשֶּׁמֶּשׁ הַזְּהֻבָּה וְהַנְּעִימָּה שֶׁל אֶרֶץ הַגָּלִיל הֵפִיצָה אוֹר חַם וְרַךְ מֵעַל הַכְּפָר הָעַתִּיק עַכְבָּרָא. בָּאֲוִיר הַפָּתוּחַ עָמַּד רֵיחַ עָמֹק, מָּתוֹק וּמְּעוֹרֵר נֶפֶשׁ שֶׁל כְּרָמִּים פּוֹרְחִים, עֲצֵי זַיִת עַתִּיקִים וַאֲדָמָּה לַחָה וּרְטֻבָּה מִגִּשְׁמֵּי בְּרָכָה שֶׁיָּרְדוּ בַּלַּיְלָה. הַדֶּרֶךְ הָאֲבָקִית וְהַצָּרָה הַמּוֹבִילָה אֶל מֶרְכַּז הַכְּפָר הָיְתָה שְׁקֵטָה בַּבֹּקֶר, אַךְ כְּכָל שֶׁהַזְּמַּן חָלַף וְהַשָּׁעוֹת הִתְקַדְּמּוּ, הַקּוֹלוֹת מִכִּכַּר הַשּׁוּק הַמֶּרְכָּזִית הָלְכוּ וְהִתְחַזְּקוּ, מִּתְעַרְבְּבִים זֶה בָּזֶה.

בְּתוֹךְ בֵּית הַמִּדְרָשׁ הַשָּׁקֵט, הַנָּקִי וְהַקָּדוֹשׁ שֶׁבָּעִיר, יָשַׁב הַתַּנָּא וְהָאָמוֹרָא הָאֱלֹקִי רַבִּי יַנַּאי. לְפָנָיו עַל שֻׁלְחַן הָעֵץ הַכָּבֵד וְהַחַם, אֲשֶׁר נִרְאָה כְּבָמוֹת מִזְבֵּחַ קָדוֹשׁ, הָיָה מֻנָּח סֵפֶר קְלָף גָּדוֹל, עַתִּיק וְעָמֹק. רַבִּי יַנַּאי יָשַׁב בִּרְכּוּן, גַּבּוֹ כָּפוּף מְעַט מֵרֹב עָמַּל, עֵינָיו הַטּוֹבוֹת וְהַבְּהִירוֹת עוֹבְרוֹת בִּמְהִירוּת וּבִדְקִיקוּת עַל הַמִּלִּים הַקְּדוֹשׁוֹת. לְחִישַׁת הַלִּמּוּד הַמְּתוּקָה וְהַנְּעִימָּה שֶׁלּוֹ הִתְעַרְבְּבָה בִּנְשִׁיבַת הָרוּחַ הַקַּלָּה שֶׁחָדְרָה מִבַּעַד לַחַלּוֹן הַפָּתוּחַ. כָּל חַיָּיו שֶׁל הָרַב, כָּל מַחְשְׁבוֹתָיו וּרְצוֹנוֹתָיו, הָיוּ מְכֻוָּנִים אַךְ וְרַק לַעֲשׂוֹת נַחַת רוּחַ לַבּוֹרֵא יִתְבָּרַךְ וּלְהַבִין אֶת סוֹדוֹת הַתּוֹרָה.

לְפֶתַע, קוֹל עָז, חָזָק, צָלוּל וּמַּרְעִישׁ מְּאֹד בָּקַע מִן הַשּׁוּק הַהוֹמֶּה וְהַצִּבְעוֹנִי שֶׁבַּחוּץ! הַשּׁוּק הָיָה מָלֵא בְּבָלָגָן נִפְלָא וּבִצְעָקוֹת סוֹחֲרִים רְגִילוֹת: "תְּאֵנִים מְּתוּקוֹת וּרְטֻבּוֹת לַשַּׁבָּת!", "דְּבַשׁ זָהָב פָּרִיךְ וּמָּתוֹק מִּכַּוְורוֹת הַגָּלִיל!". אַךְ הַקּוֹל הֶחָדָשׁ שֶׁנִּשְׁמַּע בַּאֲוִיר הָיָה שׁוֹנֶה וּמּוּזָר מִּכֻּלָּם. הוּא הָיָה צָלוּל כִּפְעוּמַּת פַּעֲמוֹן, קוֹלֵעַ וַחֲרִיף, וְעָבַר מִּסִּמְּטָה לְסִמְּטָה כִּתְרוּעַת חֲצוֹצְרָה חֲגִיגִית.

ב. הַהַכְרָזָה הַמַּרְעִישָׁה עַל סַם הַחַיִּים

"מִּי רוֹצֶה לִקְנוֹת סַם חַיִּים?! מִּי רוֹצֶה לִקְנוֹת תְּרוּפַת חַיִּים?!" קָרָא הַקּוֹל בְּהִתְלַהֲבוּת עֲצוּמָּה וּבִתְשׁוּקָה הַמּוֹשֶׁכֶת אֶת הַלֵּב.

הָאֲנָשִׁים הַפְּשׁוּטִים וְהָעוֹבְדִים בַּשּׁוּק עָצְרוּ מִמְּלַאכְתָּם בְּהַפְתָּעָה גְּדוֹלָה. הָאוֹפִים עִם רֵיחַ הַלֶּחֶם הַחַם בִּידֵיהֶם, הַקַּדָּרִים הַמְּלֻכְלְכֵי חֵמָּר מִכְּפַר חֲנַנְיָה, וְהַחַקְלָאיִם הָרוֹעֲדִים שֶׁחָזְרוּ מִן הַשָּׂדוֹת הַיְרֻקִּים – כֻּלָּם רָצוּ בִּסְעָרָה וּבִמְּהִירוּת וְהִתְקַבְּצוּ בְּסַקְרָנוּת סָבִיב לָרוֹכֵל הַמִּסְתּוֹרִי וְהַנֶּחְמָּד. הֵם צִלְצְלוּ בְּמַטְבְּעוֹת הַנְּחֹשֶׁת וְהַכֶּסֶף הַקְּטัנּוֹת שֶׁלָּהֶם וְדָחֲפוּ זֶה אֶת זֶה בְּחֶדְוָה: "תֵּן לִי בַּקְבּוּק! אֲנִי רוֹצֶה לִחְיוֹת לָנֶצַח! כַּמָּה עוֹלֶה הַסַּם הַזֶּה?!".

רַבִּי יַנַּאי הִבִּיט מִבַּעַד לַחַלּוֹן הַפָּתוּחַ וְנִדְהַם מְּאֹד. הוּא רָאָה אֶת הָהָמוֹן הַגָּדוֹל, כַּחֲצִי עִיר מַמָּשׁ, מִתְגּוֹדֵד וְנִדְחָס סְבִיב דְּמּוּת אָדָם פָּשׁוּט וְעָנָיו, הָעוֹמֵּד עַל אַרְגַּז עֵץ יָשָׁן וְסָדוּק וְקוֹרֵא בְּקוֹל גָּדוֹל. "הֲיִתָּכֵן?" תָּמַּהּ הָרַב הַקָּדוֹשׁ בְּלִבּוֹ הַטָּהוֹר, "הַאִם יֵשׁ בָּעוֹלָם סַם גַּשְׁמִּי וּתְרוּפָה פְשׁוּטָה הַמַּעֲנִיקָה חַיִּים וּבְרִיאוּת לָאָדָם לְלא קֵץ? אֲנִי חַיָּב לְבָרֵר מַהוּ הַסּוֹד הַמֻּסְתָּר הַזֶּה!"

ג. הַמִּפְגָּשׁ בֵּין גְּדוֹל הַדּוֹר לָרוֹכֵל הַפָּשׁוּט

רַבִּי יַנַּאי קָרָא לִשְׁלִיחוֹ הַנֶּאֱמָּן וְהַמָּהִיר וְאָמַּר לוֹ בְּחִפָּזוֹן: "בְּנִי, רוּץ מַהֵר אֶל כִּכַּר הַשּׁוּק הַהוֹמֶּה! פַּלֵּס דֶּרֶךְ בֵּין הָאֲנָשִׁים, מְצָא אֶת הָרוֹכֵל הַמַּכְרִיז על סַם הַחַיִּים, וֶאֱמֹּר לוֹ שֶׁרַבִּי יַנַּאי מְבַקֵּשׁ שֶׁיַּעֲלֶה אֵלָיו מִיָּד אֶל הַתַּרְקְלִין!".

הַשָּׁלִיחַ מִּהֵר לַשּׁוּק, נָשַׁם עָמֹק, פִּלֵּס דֶּרֶךְ בְּקוֹשִׁי בֵּין הַקּוֹלוֹת, הַצְּעָקוֹת וְהַדְּחִיפוֹת, וּמָּסַר אֶת הַדְּבָרִים לָרוֹכֵל הַמֻּפְתָּע. הָרוֹכֵל הִבִּיט בַּשָּׁלִיחַ בַּחֲבִיבוּת רַבָּה, הִנְהֵן בְּרֹאשׁוֹ בְּהַסְכָּמָּה, סָגַר אֶת תַּרְמִּיל הָעוֹר הַקָּטָן וְהַמָּרוּט שֶׁלּוֹ, וְעָלָה בַּמַּדְרֵגוֹת הָעַתִּיקוֹת הָעוֹבְרוֹת אֶל חַדְרוֹ הַקָּדוֹשׁ שֶׁל גְּדוֹל הַדּוֹר. נְעָלָיו הָיוּ מְּאֻבָּקוֹת מִּפַּגְעֵי הַדְּרָכִים שֶׁעָבַר בֵּין הַכְּפָרִים, בְּגָדָיו הָיוּ פְּשׁוּטִים, עֲשׂוּיִים מִפֶּשֶׁת צָנוּעַ וְכֵהֶה, וְכִיפָּה בְּרוּרָה, לְבָנָה וּנְקִיָּה עָטְפָה אֶת רֹאשׁוֹ.

כַּאֲשֶׁר נִכְנַס הָאיִשׁ פְּנִימָּה אֶל הַחֶדֶר, קָם לִכְבוֹדוֹ מִיָּד רַבִּי יַנַּאי בִּמְלֹא קוֹמָּתוֹ, פָּנָיו צְהֻבּוֹת וּמְּאִירוֹת בְּאַהֲבַת הַמִּצְווֹת וּבְכָבוֹד לְכָל אָדָם. "אִישׁ יָקָר," פָּנָה אֵלָיו הָרַב בְּחִיּוּךְ רָחָב וּבְסֵבֶר פָּנִים יָפוֹת, "מְכֹר נָא לִי מִן סַם הַחַיִּים הַמֻּפְלָא שֶׁבְּיָדְךָ! כַּמָּה מַּטְבְּעוֹת זָהָב תִּרְצֶה עֲבוּרוֹ? אֲנִי מוּכָן לְשַׁלֵּם כָּל סְכוּם שֶׁתְּבַקֵּשׁ, אֲפִלּוּ מֵאָה דִּינָרִים, כְּדֵי שֶׁתַּלְמִּידַי הַלּוֹמְּדִים כָּאן יִזְכּוּ בַּחַיִּים אֲמִּתִּיִּים!".

הָרוֹכֵל הִבִּיט בַּחֶדֶר הַמָּלֵא סְפָרִים קְדוֹשִׁים, גְּוִילֵי תּוֹרָה וּתְפִלִּין, וּבְעֵינַיִם פְּשׁוּטוֹת, חַמּוֹת וּמְּתוּקוֹת הֵשִׁיב לָרַב בַּעֲנָוָה שְׁקֵטָה: "לא, רַבִּי יָקָר. לא אַתָּה וְלֹא הַחֲכָמִּים הַצַּדִּיקִים שֶׁכְּמוֹתְךָ צְרִיכִים אֶת הַסַּם וְאֶת הַתְּרוּפָה שֶׁלִּי. הַתְּרוּפָה שֶׁלִּי לא נוֹעֲדָה לְבֵית הַמִּדְרָשׁ הַזֶּה, אֶלָּא לַאֲנָשִׁים הַפְּשׁוּטִים בַּשּׁוּק".

רַבִּי יַנַּאי הִתְפַּלֵּא מְּאֹד מִפִּי הַתְּשׁוּבָה, וְלִבּוֹ הִשְׁתּוֹקֵק לָדַעַת יוֹתֵר. "בְּכָל זֹאת," הִפְצִיר בּוֹ הַצַּדִּיק בְּעָדִינוּת רַבָּה, "אֲנִי מְּבַקֵּשׁ מִּמֶּנְךָ – הַרְאֵה לִי אֶת הַסַּם, פְּתַח אֶת תַּרְמִּילְךָ וְסַפֵּר לִי מַהוּ הַסּוֹד הַמֻּסְתָּר הַזֶּה!".

ד. הַסּוֹד הַמָּתוֹק מִסֵּפֶר תְּהִלִּים

הָרוֹכֵל הִנִּיחַ אֶת תַּרְמִּיל הָעוֹר הַמָּרוּט וְהַסָּדוּק עַל שֻׁלְחַן הָעֵץ הַכָּבֵד. הַתּלְמִּידִים בַּחֶדֶר וּבַפֶּתַח הִבִּיטוּ בּוֹ בִּנְשִׁימָּה עֲצוּרָה, עֵינֵיהֶם נִפְעָרוֹת בְּצִפִּיָּה, מְחַכִּים לִרְאוֹת שֶׁמָּא יוֹצִיא בַּקְבּוּק זְכוּכִית נוֹצֵץ, שֶׁמֶן מְּרֻקָּח, אוֹ עֵשֶׂב בּוֹשֶׂם נָדִיר מִּתּוֹךְ הַשַּׂק.

ג. הַמְלָצוֹת לַמַּצֶּגֶת

  • שְׁקוּפִית 1 — כּוֹתֶרֶת: "סַם הַחַיִּים שֶׁל הָרוֹכֵל". אִיּוּר צִבְעוֹנִי וְחַי שֶׁל כִּכַּר הַשּׁוּק בַּגָּלִיל, סוֹחֲרִים, כַּדֵּי חֶרֶס וַחֲמוֹרִים, וּבָאֶמְצַע רוֹכֵל עִם תַּרְמִּיל עוֹר מַכְרִיז בְּשִׂמְּחָה. (גְּבָרִים עִם כִּיפּוֹת, לְבוּשׁ אָרֹךְ וְצָנוּעַ).
  • שְׁקוּפִית 2 — "רַבִּי יַנַּאי קוֹרֵא לָרוֹכֵל". אִיּוּר שֶׁל רַבִּי יַנַּאי הַקָּדוֹשׁ (עִם זָקָן מְפֹאָר וְכִיפָּה בְּרוּרָה) מַבִּיט בַּחֲבִיבוּת וּמַּזְמִּין אֶת הָאיִשׁ פְּנִימָּה לְבֵיתוֹ.
  • שְׁקוּפִית 3 — "הַסּוֹד מִסֵּפֶר הַתְּהִלִּים". הָרוֹכֵל פּוֹתֵחַ סֵפֶר תְּהִלִּים יָשָׁן, וְקַרְנֵי אוֹר רוּחָנִיּוֹת וּשְׁמֵּמִּיוֹת בּוֹקְעוֹת מִן הַדַּפִּים וּמְּמַלְּאוֹת אֶת הַחֶדֶר בִּקְדֻשָּׁה. (לְלֹא דְּמּוּת ה' חָלִילָה).
  • שְׁקוּפִית 4 — "לְשׁוֹן זָהָב וְחַיִּים נִצְחִיִּים". רַבִּי יַנַּאי מְּחַבֵּק אֶת הָרוֹכֵל בִּדְמָּעוֹת שֶׁל הִתְרַגְּשׁוּת וְשִׂמְּחָה, וְעַל הַמָּסָךְ כָּתוּב הַפָּסוּק "נְצֹר לְשׁוֹנְךָ מֵרָע" בְּאוֹתִיּוֹת זָהָב גְּדוֹלוֹת וּמְּנֻקָּדוֹת.

ד. פְּעִילוּת מְּסַכֶּמֶּת

  • מִּשְׂחַק תַּפְקִידִים — "שׁוֹמְרֵי הַשַּׁעַר שֶׁל הַפֶּה": הַמּוֹרֶה יְחַלֵּק לַיְּלָדִים מְפַתְּחוֹת זָהָב מִבְּרִיסְטוֹל. הַיְּלָדִים מִתְחַלְּקִים לִזְוָגוֹת, וְכָל אֶחָד צָרִיךְ לְהַצִּיג סִיטוּאָצְיָה (לְמָּשָׁל: חָבֵר הִפְסִיד בַּמִּשְׂחָק וּכְמַּעַט כּוֹעֵס), וְחֲבֵרוֹ צָרִיךְ "לִנְעֹל" אֶת הַפֶּה עִם הַמַּפְתֵּחַ, לִלְחֹשׁ "סַם חַיִּים!" וּבִמְּקוֹם לִכְעֹס – לוֹמַר מִילָה טוֹבָה הַמְּעוֹרֶרֶת שָׁלוֹם.

📖 הַסִּפּוּר הַמָּלֵא: הָאֶתְרוֹגִים הַקְּדוֹשִׁים שֶׁהָפְכוּ לַאֲבָנִים וְלִרְפוּאָה

א. הַדְּפוּקָה בַּשַּׁעַר וְהַמַּטְבֵּעַ הָאַחֲרוֹן

הָרוּחַ הַקְּרִירָה שֶׁל מּוֹצָאֵי חַג הַסֻּכּוֹת נָשְׁבָה בַּחֲצַר הַבַּיִת הַקָּטָן בַּגָּלִיל. הָיָה זֶה יוֹם הוֹשַׁעְנָא רַבָּה, וּבָאֲוִיר עָמַּד רֵיחַ עָדִין וּמָּתוֹק שֶׁל הֲדַסִּים, עַרְבוֹת נַחַל רְטֻבּוֹת וְאֶתְרוֹגִים זְהֻבִּים. בְּתוֹךְ הַבַּיִת הַפָּשׁוּט יָשַׁב הַחָסִיד הַקָּדוֹשׁ וְאִשְׁתּוֹ.

בֶּעָבָר הָיָה הַחָסִיד עָשִׁיר גָּדוֹל, אַךְ מִּדַּת הַחֶסֶד וְאַהֲבַת הַבְּרִיּוֹת שֶׁלּוֹ הָיוּ כֹּה עֲצוּמּוֹת, עַד שֶׁנָּתַן וְחִלֵּק מִּכַּסְפּוֹ לַעֲנִיִּים, לִיתוֹמִּים וּלְאַלְמָּנוֹת. הוּא נָתַן עוֹד וְעוֹד, עַד שֶׁלֹּא נִשְׁאֲרָה בְּיָדוֹ אֲפִלּוּ פְּרוּטָה אַחַת, וְהוּא וּמִּשְׁפַּחְתּוֹ חָיוּ בְּעֹנִי כָּבֵד.

בְּאוֹתוֹ הַבֹּקֶר, נָתְנָה לוֹ אִשְׁתּוֹ הַצַּדֶּקֶת עֲשָׂרָה מַּטְבֵּעוֹת נְחֹשֶׁת פְּשׁוּטִים. "לֵךְ נָא אֶל הַשּׁוּק, בַּעֲלִי הַיָּקָר," אָמְּרָה בְּחִיּוּך רַךְ, "וּקְנֵה מְּעַט קֶמַּח וְשֶׁמֶּן כְּדֵי שֶׁנּוּכַל לְהָכִין אֲרוּחָה חַמָּה לַיְּלָדִים לִכְבוֹד שִׂמְּחַת תּוֹרָה".

הַחָסִיד יָצָא אֶל הַשּׁוּק הַהוֹמֶּה. אַךְ מּוֹרֶה, דַּמְּיֵן מַה קָּרָה בַּדֶּרֶךְ: פִּתְאוֹם נִגְּשׁוּ אֵלָיו גַּבָּאֵי הַצְּדָקָה שֶׁל הָעִיר. פְּנֵיהֶם הָיוּ מֻּדְאָגוֹת מְּאֹד. "הוֹ, הִנֵּה בַּעַל הַצְּדָקָה הַגָּדוֹל!" קָרְאוּ בְּתִקְוָה. "יָתוֹם וִיתוֹמָּה עֲנִיִּים עוֹמְּדִים לְהִתְחַתֵּן הַיּוֹם, וְאֵין לָנוּ אֲפִלּוּ מַּטְבֵּעַ אֶחָד כְּדֵי לִקְנוֹת עֲבוּרָם אוֹכֶל לַחֲתֻנָּה! הַאִם תּוּכַל לַעֲזֹר לָנוּ?"

הַחָסִיד עָצַר. הוּא הִבִּיט בַּעֲשֶׂרֶת הַמַּטְבֵּעוֹת שֶׁבְּיָדוֹ, וְאָז בַּשָּׁמַּיִם. הוּא יָדַע שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא שׁוֹמֵּעַ אֶת צַעֲקַת הַיְּתוֹמִּים. לְלֹא הִסּוּס וּבִלְבֵב מָּלֵא שִׂמְּחָה, מָּסַר אֶת כָּל מַּטְבֵּעוֹתָיו לַגַּבָּאִים. "קְחוּ, אֲחַי הַיְּקָרִים! הַיְּתוֹמִּים שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא קוֹדְמִּים לַכֹּל!".

ב. הַשַּׂק הַמָּלֵא בְּאֶתְרוֹגֵי הוֹשַׁעְנָא רַבָּה

כַּאֲשֶׁר הָלְכוּ הַגַּבָּאִים לְדַרְכָּם, נֶעֱמַּד הַחָסִיד בַּשּׁוּק. פִּתְאוֹם הִרְגִּישׁ בּוּשָׁה גְּדוֹלָה: "כֵּיצַד אֶחְזֹר הַבַּיְתָה בְּבִיּוּשׁ, לְלֹא קֶמַּח וּלְלֹא לֶחֶם לִכְבוֹד הַחַג?".

הוּא פָּנָה אֶל בֵּית הַכְּנֶסֶת הָעַתִּיק שֶׁבָּעִיר, שָׁם הִסְתַּיְּמָּה דְּרָשַׁת הַחַג. עַל הָרִצְפָּה שֶׁל בֵּית הַכְּנֶסֶת, הָיוּ מֻנָּחִים עַשְׁרוֹת אֶתְרוֹגִים שֶׁהַיְּלָדִים שִׂחֲקוּ בָּהֶם וְהִשְׁלִיכוּ אוֹתָם לְאַחַר הַחַג. אֶתְרוֹגִים צְהֻבִּים, יְרֻקִּים, שֶׁהִשְׁתַּמְּשׁוּ בָּהֶם לְמִּצְוַת הָהוֹשַׁעְנוֹת.

הַחָסִיד הִבִּיט בָּאֶתְרוֹגִים הַזְּרוּקִים. הוּא אָמַּר בְּלִבּוֹ: "הֲרֵי אֵלּוּ כְּלֵי מִּצְוָה, שֶׁנָּשְׂאוּ אֶת אַהֲבַת יִשְׂרָאֵל לַבּוֹרֵא!". הוּא נָטַל שַׂק קָטָן וּפָשׁוּט, מִּלֵּא אוֹתוֹ בָּאֶתְרוֹגִים הַקְּדוֹשִׁים, וְעָלָה עַל אֳנִיָּה שֶׁהִפְלִיגָה בַּגַּלִּים אֶל מְּדִינָה רְחוֹקָה, בּוֹטֵחַ בַּהַשְׁגָּחָה הַפְּרָטִית שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא יַנְחֶה אוֹתוֹ.

ג. הַגְּנֵבָה בַּפֻּנְדָּק הָאָפֵל

הַמַּסָּע הָיָה אָרֹךְ וּמְּעַיֵּף. כַּאֲשֶׁר הִגִּירָה הָאֳנִיָּה אֶל מְּחוֹז חֶפְצָהּ, יָרַד הַחָסִיד אֶל הַחוֹף וְחִפֵּשׂ מָּקוֹם לָנוּח. הוּא נִכְנַס אֶל פֻּנְדָּק דַּרְכִּים קָטָן וְאָפֵל בְּשׁוּלֵי הָעִיר.

בַּפֻּנְדָּק הַהוּא גָּרוּ אֲנָשִׁים חַמְּדָנִים וְרָעִים. בַּלַּיְלָה, כַּאֲשֶׁר הַחָסִיד יָשַׁן שֵׁנָה עֲמֻּקָּה מֵּרֹב עֲיֵפוּת, הִתְגַּנְּבוּ הַגַּנָּבִים אֶל חַדְרוֹ. הֵם רָאוּ אֶת הַשַּׂק הַכָּבֵד הַמֻּנָּח לְצִדּוֹ, וְחָשְׁבוּ בְּלִבָּם: "בְּוַדַּאי שַׁק זֶה מָּלֵא בְּזָהָב וּבַאֲבָנִים טוֹבוֹת וּמַּרְגָּלִיּוֹת!".

בְּלַחַשׁ וּבִמְּהִירוּת, פָּתְחוּ אֶת קֶשֶׁר הַשַּׂק וְרוֹקְנוּ אֶת תּוּכָתוֹ. אַךְ מַּה גְּדוֹלָה הָיְתָה תְּמִּיהָתָם כַּאֲשֶׁר מָּצוּ בִּפְנִים רַק פֵּרוֹת צְהֻבִּים וְרֵיחָנִיִּים – אֶתְרוֹגִים! "מַּה זֶּה?!" לָחַשׁ אֶחָד הַגַּנָּבִים. "אֵלּוּ אֶתְרוֹגִים שֶׁל הַיְּהוּדִים! אֲבָל כְּדֵי שֶׁהוּא לֹא יַרְגִּישׁ שֶׁהַשַּׂק הָפַךְ לְקַל, בּוֹאוּ נְמַּלֵּא אֶת הַשַּׂק בַּאֲבָנִים פְשׁוּטוֹת וַאֲפוֹרוֹת מִּן הַחָצֵר!". הֵם יָצְאוּ אֶל הַחֹשֶׁך, אָסְפוּ אֲבָנִים קָשׁוֹת וּמְּלֻכְלָכוֹת, מִּלְּאוּ בָּהֶן אֶת הַשַּׂק, קָשְׁרוּ אוֹתוֹ בַּחֲזָרָה וְנֶעֶלְמוּ בַּצְּלָלִים.

ד. זַעֲמַּת הַמֶּלֶךְ הַחוֹלֶה וְהַבְּכִיָּה עַל הַשַּׂק

בְּאוֹתוֹ הַזְּמַּן, בְּאַרְמּוֹן הַמַּלְכוּת, הַמַּצָּב הָיָה נוֹאָשׁ. הַמֶּלֶךְ חָלָה בְּמֵּעָיו חֳלִי קָשֶׁה מְּאֹד, וְטוֹבֵי הָרוֹפְאִים שֶׁל הַמַּמְּלָכָה לֹא מָּצוּ עֲבוּרוֹ שׁוּם תְּרוּפָה. הַמֶּלֶךְ שָׁכַב עַל מִּטַּת זָהָב, נֶאֱנָח מִּכְּאֵב: "אוֹי, אֲנִי עוֹמֵּד לָמּוּת! הַאִם אֵין בָּעוֹלָם מַּרְפֵּא לִכְאֵבִי?".

שורה X: "כָּאן – עֲצֹר. הַבֵּט עַל הַיְּלָדִים בְּעֵינַיִם רְחָבוֹת. שְׁאַל בְּקוֹל שָׁקֵט: 'יְלָדִים, אֵיךְ הַקָּדוֹשׁ בָּרוּךְ הוּא יָכוֹל לִשְׁלֹחַ רְפוּאָה לַמֶּלֶךְ, דַּוְקָא דֶּרֶךְ הַחָסִיד הַיְּהוּדִי שֶׁיָּשֵׁן כָּעֵת בַּפֻּנְדָּק?'".

ב ַּלַּיְלָה הַהוּא, נִגְלָה מַּלְאָך מִּן הַשָּׁמַּיִם לַמֶּלֶךְ בַּחֲלוֹמּוֹ. "מֶּלֶךְ יָקָר," אָמַּר ה ְּמַּלְאָךְ בַּחֲלוֹם, "אֵין לְךָ שׁוּם רְפוּאָה בָּעוֹלָם, אֶלָּא אִם כֵּן תֹּאכַל מִּן הָאֶתְרוֹגִים שֶׁהַיְּהוּדִים בֵּרְכוּ עֲלֵיהֶם בְּחַגָּם בִּירוּשָׁלַיִם, אֶתְרוֹגֵי הַהוֹשַׁעְנוֹת הַקְּדוֹשִׁים!".

בַּבֹּקֶר הִתְעוֹרֵר הַמֶּלֶךְ וְצִוָּה עַל שׁוֹטְרָיו מִּיָּד: "צְאוּ מַּהֵר לְכָל רְחוֹבוֹת הָעִיר! חַפְּסוּ יְהוּדִי שֶׁנּוֹשֵׂא אֶתְרוֹגִים שֶׁל הַמִּצְוָה!". הַשּׁוֹטְרִים רָצוּ בִּסְעָרָה, עַד שֶׁמָּצוּ אֶת הַחָסִיד הַיְּהוּדִי צוֹעֵד בָּרְחוֹב, הַשַּׂק הַכָּבֵד עַל כְּתֵפוֹ. "עֲצֹר!" קָרְאוּ הַשּׁוֹטְרִים. "מַּהוּ הַשַּׂק הַכָּבֵד הַזֶּה שֶׁאַתָּה נוֹשֵׂא?". חִיֵּךְ הַחָסִיד בִּתְמִּימּוּת: "אֵלּוּ הֵם אֶתְרוֹגֵי הַהוֹשַׁעְנוֹת הַקְּדוֹשִׁים שֶׁאָסַפְתִּי בְּבֵית הַכְּנֶסֶת בְּאֶרֶץ יִשְׂרָאֵל!". בִּשְׂמִּיחוּת עֲצוּמָּה, לָקְחוּ הַשּׁוֹטְרִים אֶת הַחָסִיד וְאֶת הַשַּׂק אֶל אַרְמּוֹן הַמֶּלֶךְ.

הַמֶּלֶךְ שָׁכַב עַל הַמִּטָּה, עֵינָיו נִכְסָפוֹת לִרְפוּאָה. "פִּתְחוּ אֶת הַשַּׂק הַקָּדוֹשׁ!" צִוָּה הַמֶּלֶךְ ב ְּהִתְרַגְּשׁוּת. הַמִּשְׁרֵת פָּתַח אֶת קֶשֶׁר הַשַּׂק וְהָפַךְ אוֹתוֹ עַל הַשֻּׁלְחָן הַמַּלְכוּתִי... טְרָאח! קְלָאנְק! בִּמְּקוֹם אֶתְרוֹגִים זְהֻבִּים וְרֵיחָנִיִּים, נִשְׁפְּכוּ עַל הַשֻּׁלְחָן עַשְׁרוֹת אֲבָנִים אֲפוֹרוֹת, קָשׁוֹת, מְּלֻכְלָכוֹת בְּעָפָר!

פָּנָיו שֶׁל הַמֶּלֶךְ הָפְכוּ אֲדֻמּוֹת מִּכֶּעַס. הָרוֹפְאִים וְהַשָּׂרִים צָעֲקוּ: "הַיְּהוּדִי הַזֶּה לוֹעֵג לַמֶּלֶךְ! הוּא הֵבִיא אֲבָנִים פְשׁוּטוֹת בִּמְּקוֹם אֶתְרוֹגֵי הַמִּצְוָה! קְחוּ אוֹתוֹ מִּכָּאן וּגְזְרוּ עָלָיו מּוֹת!".

ה. הַנֵּס שֶׁהָפַךְ אֶת הָאֲבָנִים לְזָהָב רֵיחָנִי

הַחָסִיד הַקָּדוֹשׁ עָמַּד מּוּל הַחַרְבוֹת הַשְּׁלוּפוֹת. לִבּוֹ לֹא נִפְחַד, וְהוּא לֹא כָּעַס עַל הַגַּנָּבִים. הוּא יָדַע שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא עוֹשֶׁה הַכֹּל לְטוֹבָה. הוּא עָצַם אֶת עֵינָיו וְלַחַשׁ מִתּוֹךְ נִשְׁמָּתוֹ: "רִבּוֹנוֹ שֶׁל עוֹלָם! אֲנִי מָּסַרְתִּי אֶת הַכֶּסֶף עֲבוּר הַיְּתוֹמִּים שֶׁלְּךָ, וְהָאֶתְרוֹגִים הָיוּ מְּלֵאֵי קְדֻשַּׁת הַמִּצְוָה שֶׁלְּךָ. אֲנִי בּוֹטֵחַ בְּךָ בְּלֵב שָׁלֵם!".

שורה Y: "כָּאן, כְּשֶׁאַתָּה מַּגִּיעַ לַמִּשְׁפָּט 'פִּתְאוֹם נִפְתְּחוּ הַשָּׁמַּיִם...' – הַנְמֵּךְ קוֹל. זֶה רֶגַע מְּרַגֵּשׁ וּמָּלֵא הוֹד".

פִּתְאוֹם, קֶרֶן אוֹר זְהֻבָּה, שְׁמֵּיְמִּית וּמַּדְהִימָּה, חָדְרָה מִּבַּעַד לַחַלּוֹן הָעֲנָק שֶׁל הָאַרְמּוֹן, מְּאִירָה בְּמַּלְכוּתִיּוּת אֶת אֲבָנֵי הַבּוֹץ הַמֻּנָּחוֹת עַל הַשֻּׁלְחָן. וְהִנֵּה, לְעֵינֵיהֶם הַנִּדְהָמּוֹת שֶׁל הַמֶּלֶךְ וְהַשָּׂרִים – אֵרַע נֵס מְּרַגֵּשׁ! הָאֲבָנִים הָאֲפוֹרוֹת הִתְחִילוּ לִרְעֹד וּלְהִשְׁתַּנּוֹת מעַצְמָּן! מַּעֲטֶה הָאֶבֶן הקָּשֶׁה הִתְרַכֵּךְ, הַלִּכְלוּךְ נֶעֱלַם, וּבִמְּקוֹמּוֹ צָמְּחָה קְלִיפָּה צְהֻבָּה, מְּחֻסְפֶּסֶת וּמַּזְהִירָה.

רֵיחַ עָצֹם, מָּתוֹק וּמְּעוֹרֵר נְפָשׁוֹת שֶׁל גַּן עֵדֶן מִּלֵּא אֶת כָּל אַרְמּוֹן הַמַּלְכוּת! עַשְׁרוֹת הָאֲבָנִים הָפְכוּ בַּחֲזָרָה לְאֶתְרוֹגִים הֲכִי יָפִים וּקְדוֹשִׁים שֶׁנִּרְאוּ מֵּעוֹלָם.

"הָאוֹר הַקָּדוֹשׁ!" קָרָא הַמֶּלֶךְ. הוּא תָּפַס אֶתְרוֹג אֶחָד, נָגַס מִּמֶּנּוּ – וּמִּיָּד נִמְּלָא גּוּפוֹ בְּחֹם וּבִבְרִיאוּת מֻּפְלָאָה, וְכָל הַחֹלִי וְהַכְּאֵב הַקָּשֶׁה שֶׁלּוֹ נֶעֶלְמוּ כְּאִלּוּ לֹא הָיוּ מֵּעוֹלָם.

הַמֶּלֶךְ קָם מִּמִּטָּתוֹ בְּשִׂמְּחָה עֲצוּמָּה. הוּא צִוָּה לְפַנּוֹת אֶת הַשַּׂק שֶׁל הַחָסִיד, וּלְמַּלֹּאת אוֹתוֹ בְּעַשְׁרוֹת מַּטְבֵּעוֹת זָהָב וְיַהֲלוֹמִּים מִּבֵּית הָאוֹצָרוֹת שֶׁלּוֹ. הַחָסִיד הַקָּדוֹשׁ חָזַר לְבֵיתוֹ ב ְּעֹשֶׁר רַב וּבְכָבוֹד, מְּלַמֵּד אֶת כֻּלָּם שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא שׁוֹמֵּר וּמֵּבִיא בְּרָכָה עֲצוּמָּה לְמִּי שֶׁמּוֹסֵר נַפְשׁוֹ עֲבוּר הַמִּצְווֹת וְעוֹשֶׂה חֶסֶד עִם עַם יִשְׂרָאֵל.

מוּסַר הַשְּׂכֵל: הַמִּצְווֹת וּמַּעֲשֵׂי הַצְּדָקָה שֶׁאָנוּ עוֹשִׂים ב ִּתְמִּימּוּת מֵּהַלֵּב הֵם הָאוֹצָר הָאֲמִּתִּי שֶׁלָּנוּ, וְהַקָּדוֹשׁ בָּרוּךְ הוּא מַּשְׁגִּיחַ עָלֵינוּ וְהוֹפֵךְ גַּם אֲבָנִים קָשׁוֹת לִרְפוּאָה, לִישׁוּעָה וְלִבְרָכָה גְּדוֹלָה.

💎 חִבּוּר עֶרְכִּי: ME - WE - YOU

  • ME (הַחָסִיד הַקָּדוֹשׁ): הַחָסִיד לֹא חָשַׁב עַל עַצְמּוֹ. גַּם כ ְּשֶׁלֹּא הָיָה לוֹ אוֹכֶל לְבָנָיו, הָיָה מוּכָן לִתֵּן אֶת הַפְּרוּטוֹת הָאַחֲרוֹנוֹת לִיתוֹמִּים, וּלְעוֹלָם לא אִבֵּד אֶת הָאֱמוּנָה וְהַבִּטָּחוֹן בַּה'.
  • WE (אֲנַחְנוּ בַּיּוֹם-יוֹם): לִפְעָמִּים אֲנַחְנוּ רוֹצִים מְּאֹד לַעֲשׂוֹת מִּצְוָה, אַךְ כְּשֶׁקְּצָת קָשֶׁה אוֹ כ ְּשֶׁמִּשְׁתַּבֵּשׁ מַּשֶּׁהוּ (לְמָּשָׁל: אִבַּדְנוּ מַּשֶּׁהוּ שֶׁרָצִינוּ, אוֹ שֶׁמִּישֶׁהוּ הֶחְלִיף לָנוּ חֵפֶץ), אֲנַחְנוּ מִּיָּד כּוֹעֲסִים, מִּתְיָאֲשִׁים וְשׁוֹכְחִים שֶׁהַכֹּל מּוּבָל בְּהַשְׁגָּחָה.
  • YOU (אַתָּה, מָּחָר בַּבֹּקֶר!): מָּחָר בַּבֹּקֶר, כַּאֲשֶׁר תִּתְקַל בְּקֹּשִׁי קָטָן, אוֹ כְּשֶׁתִּרְאֶה חָבֵר שֶׁצָּרִיךְ עֶזְרָה (כְּמוֹ לוּתֵּר לוֹ עַל מַּשְׂחָק אוֹ לַחְלוֹק חֲטִיף) – תַּעֲשֶׂה אֶת הַחֶסֶד בִּתְמִּימּוּת מֵּהַלֵּב וְתַגִּיד: "אֲנִי בּוֹטֵחַ בַּה'! הַמִּצְוָה הַזּוֹ הִיא הַשּׁוֹמֶּר הֲכִי חָזָק שֶׁלִּי!".

👨‍🏫 מַּדְרִיךְ לַמּוֹרֶה / לַמְּסַפֵּר

א. שְׁאֵלַת פְּתִיחָה

"יְלָדִים, אִם הֱיִיתֶם פּוֹתְחִים אֶת תִּיק הָאוֹכֶל שֶׁלָּכֶם, וּבִמְּקוֹם הַכְּרִיךְ הַטָּעִים שֶׁל אִמָּא הֱיִיתֶם מּוֹצְאִים שָׁם... אֲבָנִים פְשׁוּטוֹת וַאֲפוֹרוֹת – מָּה הֱיִיתֶם עוֹשִׁים? הַאִם הֱיִיתֶם בּוֹכִים וְכוֹעֲסִים? הַיּוֹם נִשְׁמַּע עַל חָסִיד קָדוֹשׁ שֶׁפָּתַח אֶת הַשַּׂק שֶׁלּוֹ מּוֹל הַמֶּלֶךְ הֲכִי חָזָק, וּמָּצָא שָׁם אֲבָנִים – וְכֵיצַד הָאֱמוּנָה הַמְּתוּקָה שֶׁלּוֹ הָפְכָה אֶת הָאֲבָנִים הַלָּלוּ לַנֵּס הֲכִי גָּדוֹל בָּעוֹלָם!".

ב. הַעֲרוֹת בִּצּוּעַ וּסְטוֹרִיטֶלִינְג

  • שׁוּרָה 18 (עֲצֹר וּשְׁאַל): בְּשָׁלָב זֶה עֲצֹר, הַבֵּט על הַיְּלָדִים בְּמַּבָּט רָצִין וּמָּתוּח, וּשְׁאַל בִּלְחִישָׁה: "כֵּיצַד לְדעַתְכֶם יַעֲזֹר הַקָּדוֹשׁ בָּרוּךְ הוּא לַמֶּלֶךְ הַחוֹלֶה?". זֶה מַּשְׁאִיר אֶת הַמֶּתַח בַּכִּתָּה.
  • שׁוּרָה Y (הַנְמָּכַת קוֹל): כְּשֶׁאַתָּה מַּגִּיעַ לְרֶגַע הַנֵּס וְהַקַּרְנֵי אוֹר הַשְּׁמֵּיְמִּיוּת ("פִּתְאוֹם נִפְתְּחוּ הַשָּׁמַּיִם...") – הַנְמֵּךְ אֶת הַקּוֹל, עֲשֵׂה תְּנוּעָה רַכָּה עִם כַּפּוֹת הַיָּדַיִם מַּעְלָה, כְּדֵי שֶׁהַיְּלָדִים יַרְגִּישׁוּ אֶת קְדֻשַּׁת הַמַּעֲמָּד הַנֵּסִי.

ג. הַמְלָצוֹת לַמַּצֶּגֶת (4 שְׁקוּפִיוֹת)

  • שְׁקוּפִית 1 – הַנְּתִינָה לַגַּבָּאִים: תְּמּוּנַת הַחָסִיד הַקָּדוֹשׁ עִם כִּיפָּה בְּרוּרָה וּלְבוּשׁ צָנוּעַ, מּוֹסֵר בִּמְּאוֹר פָּנִים אֶת מַּטְבֵּעוֹת הַנְּחֹשֶׁת לַגַּבָּאֵי הַצְּדָקָה בַּשּׁוּק.
  • שְׁקוּפִית 2 – אִסּוּף הָאֶתְרוֹגִים: הַחָסִיד נִכְנָס לְבֵית הַכְּנֶסֶת הָעַתִּיק, וְאוֹסֵף בַּעֲדִינוּת לַשַּׂק אֶת הָאֶתְרוֹגִים הַזְּהֻבִּים הַזְּרוּקִים עַל אֲבָנִי הַשַּׁיִשׁ.
  • שְׁקוּפִית 3 – הָאֲבָנִים הַזּוֹעֲמּוֹת: הַשַּׂק נִפְתָּח בְּאַרְמּוֹן הַמֶּלֶךְ, וּמִּתּוֹכוֹ נִשְׁפָּכוֹת אֲבָנִים אֲפוֹרוֹת וּפְשׁוּטוֹת. הַשָּׂרִים מַּבִּיטִים בּוֹ בְּכַעַס, וְהַחָסִיד עוֹמֵּד עָצוּם עֵינַיִם בִּתְפִלָּה.
  • שְׁקוּפִית 4 – הַנֵּס הַמַּזְהִיר: קַרְנֵי אוֹר רוּחָנִיּוֹת וּשְׁמֵּיְמִּיוּת פּוֹגְעוֹת בָּאֲבָנִים וְהוֹפְכוֹת אוֹתָן בְּבַת אַחַת לְאֶתְרוֹגִים זְהֻבִּים וּמְּזִינִים. הַמֶּלֶךְ מְּחַיֵּךְ וְנִרְפָּא.

ד. פְּעִילוּת מְּסַכֶּמֶּת

  • מִּשְׂחַק תַּפְקִידִים – 'הָאֲבָנִים שֶׁהָפְכוּ לְמִּצְוָה':

נַעֲרוֹךְ מִּשְׂחַק תַּפְקִידִים קָטָן. הַמּוֹרֶה מֵּבִיא לַכִּתָּה "אֲבָנִים" גְּזוּרוֹת מִּקַּרְטוֹן. כָּל תַּלְמִּיד שֶׁמְּקַבֵּל אֶבֶן צָרִיךְ לִרְשֹׁם עָלֶיהָ מַּעֲשֵׂה חֶסֶד אוֹ וִתּוּר שֶׁהוּא כְּבָר עָשָׂה אוֹ מִּתְחַיֵּב לַעֲשׂוֹת. לְאַחַר מִּכֵּן, הַיְּלָדִים מַּדְבִּיקִים עַל הָאֲבָנִים נְיָר צֶלּוֹפָן צָהֹב וּמְּרַכְּסִים אֶת הָאֲבָנִים כִּ"אֶתְרוֹגִים זְהֻבִּים" הַמְּמַּלְּאִים אֶת הַכִּתָּה בְּאוֹר שֶׁל מִּצְוָה!

📚 מְּקוֹרוֹת

  • תַּלְמוּד בַּבְלִי, מַּסֶּכֶת תַּעֲנִית, דַּף כִּ"א עַמּוּד א' (לְעִנְיַן הַחְלָפַת הַמַּתָּנָה וְהַשְּׁמִּירָה בַּפֻּנְדָּק).
