הָרַב שֶׁלֹּא הִכִּיר אֶת הָרַמְבַּ"ם הַזֶּה

📖 בפרק זה:

  • סיפור: בקשת הסכמה לספר על מצוות השמחה וההודיה בקניית בית — והתגובה הנדהמת של רב בקי ברמב"ם
  • איפה בדיוק מסתתרת המצווה שאפילו לומדים ותיקים מפספסים
"אין כזה רמב"ם" — רב שלמד רמב"ם שנים ולא ראה את זה

הַשָּׁנָה זָכִיתִי לְהוֹצִיא סֵפֶר בְּנוֹשֵׂא הַשִּׂמְחָה וְהַהוֹדָיָה בִּקְנִיַּת בַּיִת בְּאֶרֶץ יִשְׂרָאֵל. וְקָרָה לִי בַּדֶּרֶךְ סִפּוּר מְעַנְיֵן, שֶׁפֻּרְסַם לְאַחַר מִכֵּן גַּם בְּעִתּוֹן. הָלַכְתִּי לְקַבֵּל הַסְכָּמָה מֵאֶחָד מֵרַבָּנֵי הֶעָרִים בָּאָרֶץ, וְאָמַרְתִּי לוֹ: "הָרַב, הַסֵּפֶר עוֹסֵק בַּמִּצְוָה שֶׁהָרַמְבַּ"ם כּוֹתֵב, שֶׁיְּהוּדִי הַקּוֹנֶה בַּיִת צָרִיךְ שָׁנָה שְׁלֵמָה לִשְׂמֹחַ וּלְהוֹדוֹת לַקָּדוֹשׁ בָּרוּךְ הוּא עַל הַזְּכוּת הַנִּפְלָאָה שֶׁזִּכָּה אוֹתוֹ".

אוֹתוֹ רַב הָיָה בָּקִי מְאֹד בְּרַמְבַּ"ם — לֹא רַק לָמַד אוֹתוֹ, אֶלָּא גַּם לִמֵּד אוֹתוֹ בְּעַצְמוֹ, וּבִכְמוּת נִכֶּרֶת. הוּא הִבִּיט בִּי בְּתִמָּהוֹן וְאָמַר: "אֵין כָּזֶה רַמְבַּ"ם. תִּרְאֶה, לָמַדְתִּי רַמְבַּ"ם לֹא מֵהַיּוֹם וְלֹא מֵאֶתְמוֹל — הַרְבֵּה שָׁנִים אֲנִי לוֹמֵד רַמְבַּ"ם. לֹא יִתָּכֵן שֶׁיֵּשׁ כָּזֶה רַמְבַּ"ם וְלֹא רָאִיתִי אוֹתוֹ. פָּשׁוּט לֹא יִתָּכֵן".

"זֶה בְּסֵדֶר", אָמַרְתִּי לוֹ בְּשֶׁקֶט, "יֵשׁ כָּזֶה רַמְבַּ"ם". וְהַרַב, שֶׁלָּמַד בֶּאֱמֶת אֶת כָּל הָרַמְבַּ"ם, לֹא הֵבִין אֵיךְ זֶה יָכוֹל לִהְיוֹת. אֵיךְ שְׁנֵינוּ צוֹדְקִים?

אֵיפֹה זֶה כָּתוּב בֶּאֱמֶת

הַתְּשׁוּבָה הִתְבָּרְרָה מִיָּד. לָרַמְבַּ"ם יֵשׁ מִנְהָג קָבוּעַ: לִפְנֵי כָּל סֵפֶר וָסֵפֶר בְּמִשְׁנֵה תּוֹרָה, הוּא כּוֹתֵב הַקְדָּמָה, וּבָהּ הוּא מוֹנֶה אֶת הַמִּצְווֹת שֶׁיִּתְבָּאֲרוּ בְּאוֹתוֹ סֵפֶר. וּבַהַקְדָּמָה לְהִלְכוֹת מְלָכִים, הָרַמְבַּ"ם כּוֹתֵב בְּמִפְעָל שֶׁבְּאוֹתוֹ סֵפֶר יִתְבָּאֵר גַּם עִנְיַן הַמִּצְוָה שֶׁל אָדָם הַבּוֹנֶה בַּיִת, לִהְיוֹת שָׂמֵחַ שָׁנָה שְׁלֵמָה עַל שֶׁזָּכָה לִבְנוֹת בַּיִת.

"הִנֵּה, הָרַב, זֶה כָּתוּב", אָמַרְתִּי. "אֲבָל הָרַב לָמַד אֶת הַהֲלָכוֹת — לֹא אֶת הַהַקְדָּמָה. זֶה מוֹפִיעַ בַּהַקְדָּמָה, וְלֹא בְּגוּף הַהֲלָכוֹת עַצְמָן". הָרַב הִתְרַגֵּשׁ נוֹרָא מִזֶּה. הוּא אָמַר לִי כַּמָּה חָשׁוּב לְהוֹצִיא אֶת הַדָּבָר הַזֶּה לְאוֹר, כִּי כַּנִּרְאֶה רַבִּים אֵינָם יוֹדְעִים אוֹתוֹ — אֲפִלּוּ תַּלְמִידֵי חֲכָמִים וְלוֹמְדֵי רַמְבַּ"ם וָתִיקִים, שֶׁפָּשׁוּט לֹא הִתְעַכְּבוּ עַל הַהַקְדָּמוֹת.

זוֹ נְקֻדָּה שֶׁכְּדַאי לָשֵׂאת אִתָּנוּ עוֹד לִפְנֵי שֶׁנַּגִּיעַ לְגוּף הַדִּין: מִצְוֹת שְׁלֵמוֹת יְכוֹלוֹת לְהִסָּתֵר בַּהַקְדָּמוֹת שֶׁאָנוּ רְגִילִים לְדַלֵּג עֲלֵיהֶן. לְעִתִּים הָעֹשֶׁר הָאֲמִתִּי לֹא נִמְצָא בְּגוּף הַסֵּפֶר, אֶלָּא בַּשּׁוּרוֹת שֶׁלִּפְנֵי הַשַּׁעַר.

רַשִׁ"י מוּל הָרַמְבַּ"ם: כַּמָּה מְרַבֶּה "נָקִי לְבֵיתוֹ"

📖 בפרק זה:

  • הפסוק בפרשת כי תצא, על החתן הפטור מיציאה לצבא
  • שיטת רש"י: הריבוי מוגבל לפְּטוֹר מן המלחמה בלבד
  • שיטת הרמב"ם: הריבוי חל על הכל — כולל מצוות השמחה
  • ההסבר ההלכתי לביסוס המחלוקת, ומה הקשר בין פרשת כי תצא לפרשת שופטים

הַפָּסוּק בְּפָרָשַׁת כִּי תֵצֵא

הַנּוֹשֵׂא בְּעַצְמוֹ מְקוֹרוֹ אֵינוֹ בְּפָרָשַׁת שׁוֹפְטִים, אֶלָּא בְּפָרָשַׁת כִּי תֵצֵא, בַּפָּסוּק הָעוֹסֵק בְּפֵטוֹר חָתָן מִן הַצָּבָא:

"כִּי־יִקַּח אִישׁ אִשָּׁה חֲדָשָׁה, לֹא יֵצֵא בַּצָּבָא וְלֹא־יַעֲבֹר עָלָיו לְכָל־דָּבָר, נָקִי יִהְיֶה לְבֵיתוֹ שָׁנָה אֶחָת וְשִׂמַּח אֶת־אִשְׁתּוֹ אֲשֶׁר־לָקָח"

דברים, פרק כ"ד, פסוק ה'

וּמְעַנְיֵן לָשֵׂים לֵב: כְּבָר קֹדֶם לָכֵן, בְּפָרָשַׁת שׁוֹפְטִים עַצְמָהּ, מוֹפִיעַ הַכְּלָל הַכְּלָלִי — שֶׁמִּי שֶׁבָּנָה בַּיִת חָדָשׁ, נָטַע כֶּרֶם חָדָשׁ, אוֹ אֵרַשׂ אִשָּׁה, פָּטוּר מִלָּצֵאת לַמִּלְחָמָה. פָּרָשַׁת כִּי תֵצֵא בָּאָה וּמוֹסִיפָה פְּרָט מְיֻחָד עַל אוֹתוֹ כְּלָל, בְּיַחַס לַחָתָן דַּוְקָא: "נָקִי יִהְיֶה לְבֵיתוֹ שָׁנָה אֶחָת". הַזִּקָּה הַזֹּאת בֵּין שְׁתֵּי הַפָּרָשִׁיּוֹת תִּתְבָּרֵר כְּמַפְתֵּחַ לְכָל הַמַּחְלֹקֶת שֶׁלְּפָנֵינוּ.

שיטת רש"י

רַשִׁ"י מְבָאֵר עַל "נָקִי יִהְיֶה לְבֵיתוֹ": "לְרַבּוֹת בּוֹנֶה בַּיִת וַחֲנָכוֹ — אֵינוֹ זָז מִבֵּיתוֹ בִּשְׁבִיל צָרְכֵי הַמִּלְחָמָה". כְּלוֹמַר, הַמִּלָּה "לְבֵיתוֹ" בָּאָה לְרַבּוֹת רַק אֶת הַפְּטוֹר מִצָּרְכֵי צָבָא לְמִי שֶׁבָּנָה בַּיִת — וְתוּ לֹא. אֵין כָּאן שׁוּם רֶמֶז לְמִצְוַת שִׂמְחָה בְּעֶצֶם הַבַּיִת.

שיטת הרמב"ם

הָרַמְבַּ"ם סוֹבֵר שֶׁהָרִבּוּי חָל עַל הַכֹּל — גַּם עַל מִצְוַת "וְשִׂמַּח". לְשִׁיטָתוֹ, בּוֹנֶה בַּיִת מְרֻבֶּה לֹא רַק לְעִנְיַן הַפְּטוֹר, אֶלָּא גַּם לְעִנְיַן חוֹבַת הַשִּׂמְחָה עַצְמָהּ — כְּשֵׁם שֶׁחָתָן חַיָּב לִשְׂמֹחַ עִם אִשְׁתּוֹ, כָּךְ יֵשׁ חִיּוּב לִשְׂמֹחַ עַל זְכוּת הַבַּיִת.

שִׁיטַת רַשִׁ"י: רִבּוּי חֶלְקִי

רַשִׁ"י קוֹרֵא אֶת "וְשִׂמַּח אֶת אִשְׁתּוֹ" כְּמִצְוָה שֶׁכֻּלָּהּ מְכֻוֶּנֶת כְּלַפֵּי הָאִשָּׁה: יֵשׁ לְאָדָם מִצְוָה לְשַׂמֵּחַ אֶת אִשְׁתּוֹ, לְרַצּוֹתָהּ, לִהְיוֹת עִמָּהּ. וְאִם כֵּן, מִצְוָה זוֹ פָּשׁוּט אֵינָה שַׁיֶּכֶת לְבַיִת. אֵין דֶּרֶךְ לְשַׂמֵּחַ בַּיִת — הַבַּיִת דּוֹמֵם, וְאִי אֶפְשָׁר לְשַׂמֵּחַ מַשֶּׁהוּ שֶׁאֵין בּוֹ חַיִּים. לָכֵן, לְפִי רַשִׁ"י, הָרִבּוּי "נָקִי יִהְיֶה לְבֵיתוֹ" מְרַבֶּה רַק אֶת הַפְּטוֹר מִן הַצָּבָא לְמִי שֶׁבָּנָה בַּיִת — אֲבָל אֶת חוֹבַת הַשִּׂמְחָה, מַשְׁאִיר רַשִׁ"י מְכֻוֶּנֶת אֶל הָאִשָּׁה בִּלְבַד.

שִׁיטַת הָרַמְבַּ"ם: רִבּוּי לְכָל דָּבָר

הָרַמְבַּ"ם, לְעֻמַּת זֹאת, קוֹרֵא אֶת "וְשִׂמַּח" כְּ"יִשְׂמַח עִם אִשְׁתּוֹ" — כְּלוֹמַר, יִשְׂמַח יַחַד אִתָּהּ עַל הַמַּצָּב הֶחָדָשׁ שֶׁאֵלָיו הִגִּיעוּ שְׁנֵיהֶם. וְאִם הַשִּׂמְחָה הִיא עַל הַמַּצָּב הֶחָדָשׁ, מֵאֵלָיו אֶפְשָׁר לוֹמַר שֶׁגַּם הַבַּיִת נִכְלָל בָּהּ — כִּי הַשִּׂמְחָה כָּאן אֵינָהּ עַל הָאִשָּׁה בִּלְבַד, אֶלָּא עַל הַזְּכוּת הַנִּפְלָאָה שֶׁל בַּיִת חָדָשׁ בְּיִשְׂרָאֵל, שֶׁאֵלֶיהָ זָכוּ שְׁנֵיהֶם יַחַד.

וְהַבָּסִיס הַהִלְכָתִי לְהַרְחָבָה הַזּוֹ נָעוּץ בִּכְלַל דְּרָשָׁה יָדוּעַ מִשְּׁלֹשׁ עֶשְׂרֵה הַמִּדּוֹת שֶׁהַתּוֹרָה נִדְרֶשֶׁת בָּהֶן: כָּל דָּבָר שֶׁהָיָה בִּכְלָל וְיָצָא מִן הַכְּלָל לְלַמֵּד — לֹא לְלַמֵּד עַל עַצְמוֹ יָצָא, אֶלָּא לְלַמֵּד עַל הַכְּלָל כֻּלּוֹ יָצָא. וְכָאן זֶה בְּדִיּוּק הַמַּצָּב: בַּיִת, כֶּרֶם וְאִשָּׁה הָיוּ כֻּלָּם בִּכְלָל אֶחָד — כְּלַל הַפְּטוֹרִים מִן הַמִּלְחָמָה שֶׁבְּפָרָשַׁת שׁוֹפְטִים. וּפָרָשַׁת כִּי תֵצֵא בָּאָה וְהוֹצִיאָה אֶת עִנְיַן הֶחָתָן מִן הַכְּלָל הַזֶּה, כְּדֵי לְלַמֵּד עָלָיו דִּין מְיֻחָד — שָׁנָה שְׁלֵמָה שֶׁל "נָקִי לְבֵיתוֹ".

כלל שהיה בכלל ויצא מן הכלל ללמד — לא ללמד על עצמו יצא, אלא ללמד על הכלל כולו יצא

משלוש עשרה המידות שהתורה נדרשת בהן

לְפִי כְּלַל הַדְּרָשָׁה הַזֶּה, כְּשֶׁפְּרָט אֶחָד יוֹצֵא מִתּוֹךְ כְּלָל כְּדֵי לְלַמֵּד עָלָיו דִּין חָדָשׁ, אֵין הַדִּין מְצֻמְצָם רַק לְאוֹתוֹ פְּרָט — הוּא חוֹזֵר וְחָל עַל כָּל הַכְּלָל כֻּלּוֹ. וְלָכֵן, סוֹבֵר הָרַמְבַּ"ם, אֵין לְצַמְצֵם אֶת הָרִבּוּי שֶׁל "נָקִי יִהְיֶה לְבֵיתוֹ" רַק לְעִנְיָן אֶחָד — אֶלָּא הוּא חָל עַל כָּל מַה שֶּׁקָּשׁוּר לְאוֹתוֹ מַצָּב, וּבִכְלָלוֹ גַּם מִצְוַת הַשִּׂמְחָה בַּבַּיִת עַצְמוֹ. וְזֶהוּ, אַגַּב, גַּם הַהֶסְבֵּר לְכָךְ שֶׁמְּקוֹר הַדִּין הַזֶּה נִדּוֹן בְּהִלְכוֹת מְלָכִים שֶׁל הָרַמְבַּ"ם, בְּתוֹךְ "סֵפֶר שׁוֹפְטִים" שֶׁלּוֹ — שֵׁם י"ד סְפָרָיו שֶׁל מִשְׁנֵה תּוֹרָה — אַף שֶׁהַפָּסוּק עַצְמוֹ נִמְצָא בְּפָרָשַׁת כִּי תֵצֵא: הַכְּלָל הַכְּלָלִי שַׁיָּךְ לְשׁוֹפְטִים, וְהַפְּרָט הַמְּפֹרָט לְכִי תֵצֵא, וּשְׁתֵּיהֶן יַחַד בּוֹנוֹת אֶת הַדִּין.

זְכוּת שֶׁל דּוֹרוֹת

כְּדֵי לְהָבִין כַּמָּה עָמֹק דִּין הָרַמְבַּ"ם הַזֶּה, כְּדַאי לְהִתְבּוֹנֵן בְּמַה שֶּׁמִּסְתַּתֵּר מֵאֲחוֹרֵי הַמִּלִּים הַיְּבֵשׁוֹת "קְנִיַּת בַּיִת". בְּמֶשֶׁךְ מֵאוֹת שָׁנִים הָיָה אָסוּר לִיהוּדִים לִקְנוֹת בַּיִת בְּאֶרֶץ יִשְׂרָאֵל — לֹא הַתֻּרְכִּים וְלֹא הָעוֹתְמָאנִים הִרְשׁוּ זֹאת. הַיּוֹם, בָּרוּךְ הַשֵּׁם, יְהוּדִי יָכוֹל לִבְנוֹת בַּיִת, לִקְנוֹת בַּיִת, אִם יֵשׁ בְּיָדוֹ הוֹן עַצְמִי — וְזֶה בִּכְלָל לֹא מוּבָן מֵאֵלָיו. זוֹ זְכוּת שֶׁדּוֹרוֹת שְׁלֵמִים לְפָנֵינוּ לֹא זָכוּ לָהּ.

עֶשְׂרִים וַחֲמִשָּׁה אָחוּז, וְשִׁבְעִים וַחֲמִשָּׁה אָחוּז

וּבְכָל זֹאת, כְּשֶׁמַּגִּיעִים לְמַעֲשֶׂה, הַשִּׂמְחָה לֹא תָּמִיד קַלָּה. אַבְרֵךְ צָעִיר קוֹנֶה דִּירָה בְּמִיליוֹן שֶׁקֶל, נָנִיחַ. הוּא מְשַׁלֵּם עֶשְׂרִים וַחֲמִשָּׁה אָחוּז הוֹן עַצְמִי — כְּ־250,000 שֶׁקֶל — וְלוֹקֵחַ עַל עַצְמוֹ מַשְׁכַּנְתָּא שֶׁל 750,000 שֶׁקֶל, לְעֶשְׂרִים וְחָמֵשׁ שָׁנִים. סְכוּם שֶׁכָּזֶה, לְאָדָם צָעִיר, אֵינוֹ עִנְיָן פָּעוּט. וְהוּא אוֹמֵר לְעַצְמוֹ: לִשְׂמֹחַ? אֵיךְ לִשְׂמֹחַ? בַּלַּיְלָה אֲנִי מָלֵא פַּחַד — מָה לָקַחְתִּי עַל עַצְמִי, מֵאֵיפֹה אֲשַׁלֵּם אֶת זֶה, גַּם מַה שֶּׁהָיָה לִי הִפְסַדְתִּי, הַבַּנְק לוֹקֵחַ הַכֹּל, וְעוֹד רִבִּיּוֹת וְעוֹד עֲמָלוֹת.

אֵלֶּה, כָּךְ בְּדִיּוּק, דְּבָרִים שֶׁאֲנָשִׁים אוֹמְרִים בְּפֶה מָלֵא. וְעִם כֵּן, מְלַמֵּד הָרַמְבַּ"ם, יֵשׁ כָּאן סְגֻלָּה נִפְלָאָה מִן הַתּוֹרָה.

הַמַּפְתֵּחַ לְהֶמְשֵׁךְ הַשֶּׁפַע

אִם תֵּדַע לְהוֹדוֹת לַקָּדוֹשׁ בָּרוּךְ הוּא עַל מַה שֶּׁכֵּן נָתַן לְךָ — עַל אוֹתָם עֶשְׂרִים וְחָמֵשׁ אָחוּזִים שֶׁכְּבָר בְּיָדְךָ, עַל הַהוֹרִים שֶׁעָזְרוּ לְךָ וְהִגַּעְתָּ לְמָקוֹם הַזֶּה — זֶה יִהְיֶה הַכְּלִי לְהַמְשִׁיךְ עַל יָדוֹ אֶת הַשֶּׁפַע, כְּשֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא יִשְׁלַח לְךָ גַּם אֶת שִׁבְעִים וְחָמֵשׁ הָאָחוּזִים הַנּוֹתָרִים.

הַהוֹדָיָה הִיא מַפְתֵּחַ נִפְלָא. כַּאֲשֶׁר אָדָם יוֹדֵעַ לְהוֹדוֹת לַקָּדוֹשׁ בָּרוּךְ הוּא עַל מַה שֶּׁיֵּשׁ לוֹ עַד עַכְשָׁו, זֶה בְּעַצְמוֹ הַמַּפְתֵּחַ שֶׁמְּגַלְגֵּל עָלָיו אֶת כָּל הַשֶּׁפַע הָאַדִּיר שֶׁיָּבוֹא בְּעִקְבוֹתָיו. כְּשֶׁאָדָם מוֹדֶה עַל מַה שֶּׁיֵּשׁ עַד כֹּה, הוּא זוֹכֶה — וְהַקָּדוֹשׁ בָּרוּךְ הוּא מְזַכֶּה אוֹתוֹ לְעוֹד וָעוֹד. נַסּוּ וּתְרָאוּ.

"הוֹדוּ לַה' כִּי טוֹב, כִּי לְעוֹלָם חַסְדּוֹ"

תהלים, פרק קי"ח, פסוק א'

יְזַכֶּה אֶת כָּל עַם יִשְׂרָאֵל, וִיהִי רָצוֹן שֶׁהַסֵּפֶר וְהַיְּסוֹד הַזֶּה יִתְרַבּוּ בְּכָל עַם יִשְׂרָאֵל — שֶׁיִּלְמְדוּ עַל מִצְוַת הַשִּׂמְחָה, שֶׁיַּעֲשׂוּ "וַעֲדוֹת שִׂמְחָה" לְכָל מִי שֶׁבּוֹנֶה בַּיִת, שֶׁיְּדַבְּרוּ דְּרָשׁוֹת עַל הַשִּׂמְחָה וְעַל הַהוֹדָיָה, שֶׁיָּרוֹמְמוּ יַחַד אֶת הַשִּׂמְחָה וְאֶת הַהוֹדָיָה — וְיִהְיֶה שֶׁפַע טוֹב לְכָל בֵּית יִשְׂרָאֵל בְּמֵהֵרָה בְיָמֵינוּ.
