מִצְוַת הַמֶּלֶךְ: שְׁנֵי סִפְרֵי תּוֹרָה

📖 בפרק זה:

"וְכָתַב לוֹ... וְהָיְתָה עִמּוֹ"

פָּרָשַׁת שׁוֹפְטִים עוֹסֶקֶת בְּהֶרְחָבָה בְּדִינֵי הַמֶּלֶךְ, וּבֵין הַמִּצְווֹת הַמֻּטָּלוֹת עָלָיו מוֹפִיעָה מִצְוָה מְיֻחֶדֶת:

"וְהָיָה כְשִׁבְתּוֹ עַל כִּסֵּא מַמְלַכְתּוֹ, וְכָתַב לוֹ אֶת מִשְׁנֵה הַתּוֹרָה הַזֹּאת עַל סֵפֶר... וְהָיְתָה עִמּוֹ וְקָרָא בוֹ כָּל יְמֵי חַיָּיו, לְמַעַן יִלְמַד לְיִרְאָה אֶת ה' אֱלֹקָיו"

דברים, פרק י"ז, פסוקים י"ח–י"ט

הַמֶּלֶךְ מְצֻוֶּה לִכְתֹּב לוֹ סֵפֶר תּוֹרָה, וְלִקְרֹא בּוֹ כָּל יְמֵי חַיָּיו, כְּדֵי שֶׁיִּלְמַד מִמֶּנּוּ יִרְאַת שָׁמַיִם. וְחֲזַ"ל מוֹסִיפִים פְּרָט מַפְתִּיעַ: לַמֶּלֶךְ הָיוּ, לְמַעֲשֶׂה, שְׁנֵי סִפְרֵי תּוֹרָה — אֶחָד הָיָה מֻנָּח בְּבֵית גְּנָזָיו, וְהַשֵּׁנִי הָיָה קָשׁוּר אֵלָיו תָּמִיד. לְכָל מָקוֹם שֶׁהָלַךְ, הָיָה נִלְוֶה אֵלָיו סֵפֶר תּוֹרָה זֶה.

חִיּוּב כָּפוּל, מֵעֵבֶר לַחִיּוּב הָרָגִיל

וְכָאן נִשְׁאֶלֶת שְׁאֵלָה: כָּל יְהוּדִי מְצֻוֶּה לִכְתֹּב לְעַצְמוֹ סֵפֶר תּוֹרָה אֶחָד. לַמֶּלֶךְ, לְעֻמַּת זֹאת, יֵשׁ חִיּוּב לִכְתֹּב שְׁנֵי סִפְרֵי תּוֹרָה — יוֹתֵר מִן הַמִּנְהָג הָרָגִיל. מָה הַהֶבְדֵּל בֵּין מֶלֶךְ יִשְׂרָאֵל לְבֵין כָּל אֶחָד מֵאִתָּנוּ, שֶׁבִּגְלָלוֹ הוּא זָקוּק לְמִצְוָה כְּפוּלָה?

אֶפְשָׁר לְבָאֵר אֶת הָעִנְיָן בְּדֶרֶךְ הָרֶמֶז, עַל פִּי יְסוֹד יָדוּעַ בְּסֵפֶר "חוֹבוֹת הַלְּבָבוֹת". הַיְּסוֹד הַזֶּה יִהְיֶה מַפְתֵּחַ לְכָל הַפֶּרֶק הַבָּא.

לְפִי הַחֶסֶד, כָּךְ הָעֲבוֹדָה

📖 בפרק זה:

יְסוֹד חוֹבוֹת הַלְּבָבוֹת

בְּסֵפֶר "חוֹבוֹת הַלְּבָבוֹת" מְבֹאָר יְסוֹד חָשׁוּב בְּיַחַס שֶׁבֵּין הַחֶסֶד שֶׁאָדָם מְקַבֵּל לְבֵין הָעֲבוֹדָה הַמֻּטֶּלֶת עָלָיו. הַכְּלָל הוּא: מִי שֶׁמְּקַבֵּל טוֹבָה גְּדוֹלָה יוֹתֵר מֵאֵת הַקָּדוֹשׁ בָּרוּךְ הוּא, מְחֻיָּב לֹא רַק לְהוֹדוֹת עַל כָּךְ — אֶלָּא גַּם לַעֲבֹד אֶת ה' בְּמִדָּה גְּדוֹלָה יוֹתֵר מֵאֲשֶׁר אֲנָשִׁים רְגִילִים. הַמִּינִימוּם הַמֻּטָּל עַל כֻּלָּם הוּא הַתּוֹרָה, הַיְּסוֹדוֹת הַבְּסִיסִיִּים. אֲבָל מִי שֶׁהִצְטַיֵּן, מִי שֶׁזָּכָה לְמַשֶּׁהוּ מְיֻחָד — מְחֻיָּב בְּהוֹדָיָה נוֹסֶפֶת, וּבַעֲבוֹדָה נוֹסֶפֶת, בְּהֶתְאֵם לְגֹדֶל הַחֶסֶד שֶׁקִּבֵּל.

הֶדְיוֹט, כֹּהֵן גָּדוֹל וּמֶלֶךְ — קִידָה בַּתְּפִלָּה

הַגְּמָרָא מְלַמֶּדֶת אֶת הַיְּסוֹד הַזֶּה בְּצוּרָה מוּחָשִׁית מְאֹד, דֶּרֶךְ הִלְכוֹת קִידָה בִּתְפִלַּת שְׁמוֹנֶה עֶשְׂרֵה. אָדָם רָגִיל כּוֹרֵעַ בִּתְחִלַּת בִּרְכַּת "אָבוֹת" וּבְסוֹפָהּ, וּבִבְרָכַת "מוֹדִים" — סַךְ הַכֹּל אַרְבַּע כְּרִיעוֹת בְּכָל תְּפִלָּה. כֹּהֵן גָּדוֹל, לְעֻמַּת זֹאת, כּוֹרֵעַ בִּתְחִלַּת כָּל בְּרָכָה וּבְסוֹפָהּ — בְּכָל שְׁמוֹנֶה עֶשְׂרֵה הַבְּרָכוֹת. וְהַמֶּלֶךְ עוֹד יוֹתֵר: כֵּיוָן שֶׁכָּרַע פַּעַם אַחַת בִּתְחִלַּת הַתְּפִלָּה, אֵינוֹ זוֹקֵף עוֹד עַד סוֹפָהּ — הוּא נִשְׁאָר בְּמַצַּב שֶׁל הִשְׁתַּחֲוָיָה כָּל שְׁמוֹנֶה עֶשְׂרֵה כֻּלָּהּ.

"מלך כיון שכרע — שוב אינו זוקף"

גמרא, מסכת ברכות, דף ל"ד עמוד ב'

מַדּוּעַ יֵשׁ הַדְרָגָה כָּזוֹ? כִּי כָּל מִי שֶׁקִּבֵּל חֲסָדִים גְּדוֹלִים יוֹתֵר, זָקוּק לְהַכְנָעָה גְּדוֹלָה יוֹתֵר. כֹּהֵן גָּדוֹל, שֶׁזָּכָה לִהְיוֹת כֹּהֵן גָּדוֹל וּלְשָׁרֵת בַּקֹּדֶשׁ, כּוֹרֵעַ יוֹתֵר. וּמֶלֶךְ, שֶׁחֶסְדּוֹ הוּא הַגָּדוֹל מִכֻּלָּם, כּוֹרֵעַ כָּל הַזְּמַן. וְאוֹתוֹ יְסוֹד עַצְמוֹ מְלַמֵּד חוֹבוֹת הַלְּבָבוֹת בְּעִנְיַן הָעֲבוֹדָה בִּכְלָל: לְפִי הַחֶסֶד — כָּךְ הָעֲבוֹדָה.

הַסַּכָּנָה שֶׁבְּמַדְרֵגָה גְּבוֹהָה

וְיֵשׁ כָּאן גַּם צַד שֵׁנִי, פָּחוֹת נָעִים. חוֹבוֹת הַלְּבָבוֹת מְבָאֵר שֶׁמִּי שֶׁקִּבֵּל חֶסֶד גָּדוֹל וְאֵינוֹ עוֹמֵד בָּעֲבוֹדָה הַמַּתְאִימָה לוֹ — מַפְסִיד לֹא רַק אֶת מַה שֶּׁהָיָה יָכוֹל לְהַרְוִיחַ, אֶלָּא נוֹפֵל בְּמִדָּה חֲמוּרָה יוֹתֵר מִכָּל אָדָם אַחֵר. הוּא נֶעֱנָשׁ עַל דְּבָרִים דַּקִּים בְּיוֹתֵר, שֶׁעַל אָדָם רָגִיל לֹא הָיוּ נֶחְשָׁבִים כְּלָל לְעֵבֵרָה. הוּא סָטָה קְצָת מִן הַדֶּרֶךְ — דָּבָר שֶׁאֵצֶל אָדָם אַחֵר אַף לֹא הָיָה נֶחְשָׁב לְפֶגַע — וְנֶעֱנָשׁ עָלָיו, כִּי הוּא מְחֻיָּב בְּתוֹרָה גְּבוֹהָה יוֹתֵר מֵאֲנָשִׁים רְגִילִים, וַאֲפִלּוּ עַל מָה שֶׁלֹּא כָּתוּב בְּפֵרוּשׁ בַּתּוֹרָה.

וְזֶהוּ בְּדִיּוּק הָרֶמֶז שֶׁבְּשְׁנֵי סִפְרֵי הַתּוֹרָה שֶׁל הַמֶּלֶךְ: כֵּיוָן שֶׁזָּכָה לְחֶסֶד עֲצוּם שֶׁל מַלְכוּת, יֵשׁ לוֹ, כִּבְיָכוֹל, "תּוֹרָה חֲדָשָׁה" — עוֹד עֲבוֹדָה, עוֹד הוֹדָיָה שֶׁמֻּטֶּלֶת עָלָיו וְלֹא עַל אַחֵרִים. סֵפֶר תּוֹרָה נוֹסָף, כְּנֶגֶד מַדְרֵגָה נוֹסֶפֶת שֶׁל אַחֲרָיוּת.

שָׁאוּל וְדָוִד: חֵטְא אֶחָד מוּל שְׁנַיִם

הַקּוּשְׁיָה מִן הַגְּמָרָא

הַיְּסוֹד הַזֶּה מְסַיֵּעַ לְיַשֵּׁב קֻשְׁיָה יְדוּעָה שֶׁמַּעֲלָה הַגְּמָרָא. שָׁאוּל הַמֶּלֶךְ נִכְשַׁל בְּחֵטְא אֶחָד — וְאִבֵּד אֶת הַמְּלוּכָה מִבֵּיתוֹ. דָּוִד הַמֶּלֶךְ, לְעֻמָּתוֹ, נִכְשַׁל בִּשְׁנֵי חֲטָאִים חֲמוּרִים לְכָל הַדֵּעוֹת — וּבְכָל זֹאת לֹא אִבֵּד אֶת מַלְכוּתוֹ, וְהִיא נִמְשְׁכָה בְּבֵיתוֹ עַד יְמוֹת הַמָּשִׁיחַ. הֵיאַךְ יִתָּכֵן שֶׁמִּי שֶׁחָטָא פָּחוֹת אִבֵּד יוֹתֵר?

מִצְוָה שֶׁרַק אַתָּה יָכוֹל לְקַיֵּם

יֵשׁ מְתָרְצִים תֵּרוּץ יָפֶה, הַמּוּבָא בְּסֵפֶר "אִגֶּרֶת הַטִּיּוּל" — מְיֻחָס לְאֶחָד מִגְּדוֹלֵי הַמּוּסָר בְּדוֹרוֹת קוֹדְמִים. (שם המחבר המדויק, כפי שהובא בשיעור בעל פה, לא נקלט בבירור מספיק כדי לצטטו — ראו הערה בסוף.) הַיְּסוֹד שֶׁלּוֹ: יֵשׁ מִצְווֹת שֶׁכָּל אָדָם אַחֵר יָכוֹל לְקַיֵּם בִּמְקוֹמְךָ. אִם אַתָּה לֹא עוֹשֶׂה אוֹתָן, זֶה אָמְנָם פְּגָם — אֲבָל לֹא פְּגָם חָמוּר בְּיוֹתֵר, כִּי הַמִּצְוָה הִיא בַּת קִיּוּם בְּלָאו הָכִי, עַל יְדֵי מִישֶׁהוּ אַחֵר. אֲבָל יֵשׁ מִצְווֹת שֶׁרַק אַתָּה, וְאַף אֶחָד אַחֵר בָּעוֹלָם, יָכוֹל לְקַיְּמָן. אִם לֹא קִיַּמְתָּ אוֹתָן — זֶה פֶּגַע חָמוּר בְּהַרְבֵּה יוֹתֵר.

הַצַּו שֶׁשְּׁמוּאֵל הַנָּבִיא נָתַן לְשָׁאוּל — לְהַמְתִּין לוֹ קֹדֶם הַקָּרְבָּן — הָיָה מִצְוָה שֶׁרַק שָׁאוּל יָכוֹל הָיָה לְקַיְּמָהּ. אִישׁ אַחֵר לֹא הָיָה נִדְרָשׁ לְהַמְתִּין בִּמְקוֹמוֹ. כֵּיוָן שֶׁלֹּא שָׁמַע וְלֹא הִמְתִּין, הַפֶּגַע הָיָה חָמוּר בְּיוֹתֵר, כִּי הוּא — וְרַק הוּא — הָיָה יָכוֹל לְקַיֵּם אֶת הַצַּו הַזֶּה. אֵצֶל דָּוִד, לְעֻמַּת זֹאת, הַחֲטָאִים לֹא הָיוּ דְּבָרִים יְחוּדִיִּים שֶׁרַק לוֹ הָיְתָה בָּהֶם אַחֲרָיוּת — הֵם הָיוּ מִסּוּג הַנִּסְיוֹנוֹת שֶׁעוֹמְדִים בִּפְנֵי כָּל אָדָם.

הַמֶּלֶךְ הָרִאשׁוֹן

וְאֶפְשָׁר לְהוֹסִיף עַל כָּךְ, לְפִי הַיְּסוֹד שֶׁלָּמַדְנוּ בַּפֶּרֶק הַקּוֹדֵם: שָׁאוּל הַמֶּלֶךְ הָיָה הַמֶּלֶךְ הָרִאשׁוֹן שֶׁל עַם יִשְׂרָאֵל. זוֹ זְכוּת יְחִידָה בְּמִינָהּ — כָּל מֶלֶךְ שֶׁבָּא אַחֲרָיו הָיָה הֶמְשֵׁךְ שֶׁל שַׁלְשֶׁלֶת: בְּבֵית דָּוִד, לְפִי הַיֻּחֲסִין; וְאֵצֶל מַלְכֵי יִשְׂרָאֵל, בְּצוּרָה אַחֶרֶת — אֲבָל בְּכָל מִקְרֶה, הֶמְשֵׁךְ שֶׁל מַלְכוּת שֶׁכְּבָר הָיְתָה קַיֶּמֶת. שָׁאוּל לְבַדּוֹ קִבֵּל אֶת הַחֶסֶד לִהְיוֹת רִאשׁוֹן, לִפְתֹּחַ אֶת הַמַּלְכוּת מֵאַיִן. וּכְפִי שֶׁלָּמַדְנוּ — לְפִי הַחֶסֶד, כָּךְ הָעֲבוֹדָה. חֶסֶד יְחוּדִי שֶׁכָּזֶה מְחַיֵּב עֲמִידָה גְּבוֹהָה יוֹתֵר. וּכְשֶׁלֹּא עָמַד בּוֹ, אִבֵּד אֶת הַכֹּל.

זֶהוּ עִנְיָן שֶׁכְּדַאי לְכָל אֶחָד לָשֵׂאת אִתּוֹ: לִהְיוֹת עֵרָנִי, לַחְשֹׁב עַל הַחֲסָדִים שֶׁהוּא מְקַבֵּל מֵהַשֵּׁם, וְלֹא לְהִתְעַלֵּם מֵהֶם. הֵם מְחַיְּבִים אוֹתָנוּ. אָדָם בָּרִיא, אָדָם שֶׁיֵּשׁ לוֹ יְלָדִים, אָדָם נָשׂוּי — יֵשׁ לוֹ דְּבָרִים שֶׁאֵין לְאֲחֵרִים. וְכָל מַה שֶּׁיֵּשׁ לְאָדָם יוֹתֵר מִמַּה שֶּׁיֵּשׁ לַאֲחֵרִים, זֶה מְחַיֵּב אוֹתוֹ בַּעֲבוֹדָה בְּהֶתְאֵם — לַעֲשׂוֹת אֶת דִּין הַשֵּׁם, וְעוֹד דְּבָרִים לִפְנִים מִשּׁוּרַת הַדִּין, כְּדֵי לְהַרְאוֹת לַקָּדוֹשׁ בָּרוּךְ הוּא אֶת הַכָּרַת הַטּוֹב שֶׁיֵּשׁ לָנוּ עַל הַחֲסָדִים שֶׁהוּא מַשְׁפִּיעַ עָלֵינוּ.
